// @runtime VectorBase VectorStorage Vector VectorBoxing vectorWritable vectorStorageView
struct VectorBase {
    mutable Ptr<VectorBase> nativeCache;
    virtual ~VectorBase() = default;
    virtual std::span<const Value> items() const = 0;
    virtual std::size_t size() const = 0;
    virtual bool native() const = 0;
};
template <typename T> struct VectorBoxing {
    mutable Values boxed;
    mutable bool boxedReady = false;
};
template <> struct VectorBoxing<Value> {};
template <typename T> struct VectorStorage final : VectorBase {
    // The string's inline buffer avoids a separate allocation for short byte vectors.
    using Storage = std::conditional_t<std::is_same_v<T,Byte>,std::string,std::vector<T>>;
    Storage values;
    Ptr<VectorBase> owner;
    std::span<const T> view;
    [[no_unique_address]] VectorBoxing<T> boxing;
    static Storage storage(std::vector<T>&& data) {
        if constexpr (std::is_same_v<T,Byte>)
            return data.empty() ? std::string{} : std::string(reinterpret_cast<const char*>(data.data()),data.size());
        else return std::move(data);
    }
    explicit VectorStorage(std::vector<T> data) : values(storage(std::move(data))) {}
    explicit VectorStorage(std::string data) requires (std::is_same_v<T,Byte>) : values(std::move(data)) {}
    VectorStorage(const VectorStorage& other) : values(other.scalars().begin(),other.scalars().end()) {}
    std::span<const T> scalars() const {
        if (owner) return view;
        if constexpr (std::is_same_v<T,Byte>) return {reinterpret_cast<const Byte*>(values.data()),values.size()};
        else return std::span<const T>(values);
    }
    std::size_t size() const override { return scalars().size(); }
    bool native() const override { return !std::is_same_v<T,Value>; }
    std::span<const Value> items() const override {
        if constexpr (std::is_same_v<T,Value>) return scalars();
        else {
            if (!boxing.boxedReady) {
                boxing.boxed.reserve(size());
                for (const T& item : scalars()) boxing.boxed.emplace_back(item);
                boxing.boxedReady = true;
            }
            return boxing.boxed;
        }
    }
    void invalidate() {
        nativeCache.reset();
        if constexpr (!std::is_same_v<T,Value>) { boxing.boxed.clear();boxing.boxedReady=false; }
    }
};
using Vector = VectorStorage<Value>;
template <typename T> inline Ptr<VectorStorage<T>> vectorWritable(Ptr<VectorBase> base,bool reuse) {
    auto* typed = dynamic_cast<VectorStorage<T>*>(base.get());
    Ptr<VectorStorage<T>> result;
    if (typed && reuse && base.use_count() == 1 && !typed->owner)
        result = std::static_pointer_cast<VectorStorage<T>>(std::move(base));
    else if (typed) result = std::make_shared<VectorStorage<T>>(*typed);
    else {
        std::vector<T> data; data.reserve(base->size());
        for (const auto& item : base->items()) {
            if constexpr (std::is_same_v<T,Value>) data.push_back(item);
            else data.push_back(as<T>(item));
        }
        result = std::make_shared<VectorStorage<T>>(std::move(data));
    }
    result->invalidate();return result;
}
template <typename T> inline Ptr<VectorStorage<T>> vectorWritable(Value value,bool reuse) {
    return vectorWritable<T>(std::move(std::any_cast<Ptr<VectorBase>&>(value)),reuse);
}
template <typename T> inline Ptr<VectorStorage<T>> vectorStorageView(Ptr<VectorStorage<T>> source,std::size_t start,std::size_t count) {
    auto result=std::make_shared<VectorStorage<T>>(std::vector<T>{});
    result->owner=source->owner?source->owner:source;
    result->view=source->scalars().subspan(start,count);
    return result;
}

template <typename T> inline Ptr<VectorBase> scalarVector(Values data) {
    std::vector<T> result; result.reserve(data.size());
    for (const auto& item : data) result.push_back(as<T>(item));
    return scalarVector<T>(std::move(result));
}
template <typename T> inline Ptr<ScalarVector<T>> scalarData(const Ptr<VectorBase>& base) {
    if (auto typed = std::dynamic_pointer_cast<ScalarVector<T>>(base)) return typed;
    if (auto typed = std::dynamic_pointer_cast<ScalarVector<T>>(base->nativeCache)) return typed;
    std::vector<T> data; data.reserve(base->size());
    for (const auto& item : base->items()) data.push_back(as<T>(item));
    auto result = std::make_shared<ScalarVector<T>>(std::move(data));
    base->nativeCache = result;
    return result;
}
template <typename T> inline Ptr<ScalarVector<T>> scalarData(const Value& value) { return scalarData<T>(as<Ptr<VectorBase>>(value)); }
template <typename T, typename Source> inline Ptr<VectorBase> scalarPush(Source source, T item, bool reuse) {
    auto result = vectorWritable<T>(std::move(source),reuse);
    result->values.push_back(item);
    return Ptr<VectorBase>(std::move(result));
}
template <typename T, typename Source, typename Other>
inline Ptr<VectorBase> scalarConcat(Source source, const Other& other, bool reuse) {
    auto result = vectorWritable<T>(std::move(source),reuse);
    auto right = scalarData<T>(other);
    result->values.insert(result->values.end(),right->scalars().begin(),right->scalars().end());
    return Ptr<VectorBase>(std::move(result));
}
// @runtime NativeVectorReader
template <typename T> class NativeVectorReader {
    std::span<const T> scalars;
    std::span<const Value> boxed;
public:
    template <typename Source>
    explicit NativeVectorReader(const Source& source) {
        const auto* base=as<Ptr<VectorBase>>(source).get();
        if constexpr (std::is_same_v<T,Int> || std::is_same_v<T,Byte> || std::is_same_v<T,double>) {
            if (const auto* typed=dynamic_cast<const ScalarVector<T>*>(base)) {
                scalars=typed->scalars();
                return;
            }
        }
        boxed=base->items();
    }
    std::size_t size() const { return scalars.empty() ? boxed.size() : scalars.size(); }
    const T& operator[](std::size_t index) const {
        return scalars.empty() ? as<T>(boxed[index]) : scalars[index];
    }
};
// @runtime scalarContains
template <typename T, typename Source>
bool scalarContains(const Source& source, T item) {
    const NativeVectorReader<T> values(source);
    for (std::size_t i=0; i<values.size(); ++i) if (values[i]==item) return true;
    return false;
}

// @runtime vectorView
template <typename Source>
inline Ptr<VectorBase> vectorView(const Source& value, std::size_t start, std::size_t count) {
    auto source=as<Ptr<VectorBase>>(value);
    auto boxed=std::dynamic_pointer_cast<Vector>(source);
    if(boxed)return Ptr<VectorBase>(vectorStorageView<Value>(std::move(boxed),start,count));
    auto result=std::make_shared<Vector>(Values{});
    result->view=source->items().subspan(start,count);
    result->owner=std::move(source);
    return Ptr<VectorBase>(std::move(result));
}

// @runtime AdtValue
// Leave room for the variant's discriminator in the inline storage.
template <typename C> using AdtAlternative = std::conditional_t<
    sizeof(C) <= adtInlineBytes - alignof(void*) && alignof(C) <= alignof(void*) &&
        std::is_nothrow_move_constructible_v<C>, C, Ptr<C>>;

template <std::size_t Family, typename... Constructors>
class AdtValue {
    std::variant<AdtAlternative<Constructors>...> value;
    template <typename C> static AdtAlternative<C> place(C&& constructor) {
        if constexpr (std::is_same_v<C,AdtAlternative<C>>) return std::move(constructor);
        else return std::make_shared<C>(std::move(constructor));
    }
    template <typename C> static const C& unpack(const C& constructor) { return constructor; }
    template <typename C> static const C& unpack(const Ptr<C>& constructor) { return *constructor; }
public:
    template <typename C>
    explicit AdtValue(C constructor) : value(std::in_place_type<AdtAlternative<C>>,place(std::move(constructor))) {}
    template <typename C> const C* get() const {
        const auto* constructor = std::get_if<AdtAlternative<C>>(&value);
        return constructor ? &unpack(*constructor) : nullptr;
    }
    AdtTag tag() const { return {Family,value.index()}; }
    std::size_t count() const {
        static constexpr std::size_t sizes[] = {Constructors::count...};
        return sizes[value.index()];
    }
    Value field(std::size_t index) const {
        return std::visit([&](const auto& constructor) -> Value {
            Value result;
            std::size_t position = 0;
            std::apply([&](const auto&... fields) {
                ((position++ == index ? static_cast<void>(result = fields) : void()), ...);
            },unpack(constructor).fields);
            return result;
        },value);
    }
    const AdtInfo& info() const {
        static constexpr AdtInfo alternatives[] = {Constructors::info...};
        return alternatives[value.index()];
    }
};
// @runtime Adt
// A fixed-size carrier lets recursive fields and erased call sites share the
// same representation while each concrete variant retains typed fields.
class Adt {
    struct Operations {
        void (*destroy)(void*) noexcept;
        void (*copy)(void*, const void*);
        void (*move)(void*, void*) noexcept;
        AdtTag (*tag)(const void*);
        std::size_t (*count)(const void*);
        Value (*field)(const void*, std::size_t);
        const AdtInfo& (*info)(const void*);
    };
    alignas(void*) std::byte storage[adtInlineBytes];
    const Operations* ops = nullptr;

    template <typename Family> static const Family& stored(const void* source) {
        return *std::launder(static_cast<const Family*>(source));
    }
    template <typename Family> static const Operations& operations() {
        static constexpr Operations table{
            [](void* source) noexcept { std::destroy_at(std::launder(static_cast<Family*>(source))); },
            [](void* target,const void* source) { std::construct_at(static_cast<Family*>(target),stored<Family>(source)); },
            [](void* target,void* source) noexcept {
                auto* value = std::launder(static_cast<Family*>(source));
                std::construct_at(static_cast<Family*>(target),std::move(*value));
                std::destroy_at(value);
            },
            [](const void* source) { return stored<Family>(source).tag(); },
            [](const void* source) { return stored<Family>(source).count(); },
            [](const void* source,std::size_t index) { return stored<Family>(source).field(index); },
            [](const void* source) -> const AdtInfo& { return stored<Family>(source).info(); },
        };
        return table;
    }
#ifdef _MSC_VER
    // Keep ADT moves out of optimized coroutine bodies so MSVC can emit their required tail transfers.
    __declspec(noinline)
#endif
    void moveFrom(Adt& other) noexcept {
        ops = std::exchange(other.ops,nullptr);
        if (ops) ops->move(storage,other.storage);
    }
public:
    Adt() noexcept = default;
    Adt(const Adt& other) {
        if (other.ops) {
            other.ops->copy(storage,other.storage);
            ops = other.ops;
        }
    }
    Adt(Adt&& other) noexcept { moveFrom(other); }
    ~Adt() { reset(); }
    Adt& operator=(Adt other) noexcept {
        reset();
        moveFrom(other);
        return *this;
    }
    void reset() noexcept {
        if (ops) ops->destroy(storage);
        ops = nullptr;
    }
    template <typename Family, typename Constructor>
    static Adt make(Constructor constructor) {
        static_assert(sizeof(Family) <= adtInlineBytes && alignof(Family) <= alignof(void*));
        static_assert(std::is_nothrow_move_constructible_v<Family>);
        Adt result;
        std::construct_at(reinterpret_cast<Family*>(result.storage),std::move(constructor));
        result.ops = &operations<Family>();
        return result;
    }
    template <typename Family, typename C> const C* get() const {
        return ops == &operations<Family>() ? stored<Family>(storage).template get<C>() : nullptr;
    }
    AdtTag tag() const { return ops->tag(storage); }
    std::size_t count() const { return ops->count(storage); }
    Value field(std::size_t index) const { return ops->field(storage,index); }
    const AdtInfo& info() const { return ops->info(storage); }
};

// @runtime adtField
template <typename Family, typename Constructor, std::size_t Index>
auto adtField(const Adt& value) {
    if (const auto* constructor = value.template get<Family,Constructor>())
        return std::get<Index>(constructor->fields);
    auto boxed = value.field(Index);
    using Field = std::tuple_element_t<Index,decltype(Constructor::fields)>;
    if constexpr (std::is_same_v<Field,Value>) return boxed;
    else return takeAs<Field>(boxed);
}

// @runtime rangeEmpty rangeContains
inline bool rangeEmpty(const Range& r) { return r.first==r.last && !r.inclusive; }
inline bool rangeContains(const Range& r, Int item) {
    if (item==r.last) return r.inclusive;
    return r.first<=r.last ? r.first<=item && item<r.last : r.last<item && item<=r.first;
}

// @runtime setOp
template <int Operation, typename Source> inline auto setOp(Source source,const Value& key) {
    auto old=takeAs<Ptr<HashSetBase>>(source);
    if constexpr(Operation==2)return old->contains(key);
    else {
        auto result=old.use_count()==1?std::move(old):old->clone();
        if constexpr(Operation==0)result->insertKey(key);else result->erase(key);
        return result;
    }
}
// @runtime Cell Env takeCell NativeCell takeLocal
struct Cell {
    Value value; bool mutable_; bool borrowed = false;
    explicit Cell(Value data={},bool mut=false):value(std::move(data)),mutable_(mut){}
    virtual ~Cell()=default;
    virtual Value read() const{return value;}
    virtual Value take(){Value result=std::move(value);value.reset();return result;}
    virtual void write(Value data){value=std::move(data);}
};
template<typename T> struct NativeCell final : Cell {
    T value;
    explicit NativeCell(T data, bool mut = true):Cell({},mut),value(std::move(data)){}
    Value read() const override{return value;}
    Value take() override{Value result=std::move(value);value=T{};return result;}
    void write(Value data) override{
        if constexpr(std::is_same_v<T,Value>)value=std::move(data);
        else value=std::move(std::any_cast<T&>(data));
    }
};
template<typename T> T takeLocal(const Ptr<NativeCell<T>>& cell){
    if(cell.use_count()!=1||cell->borrowed)return cell->value;
    T result=std::move(cell->value);cell->value=T{};return result;
}
inline Value takeCell(const Ptr<Cell>& cell) {
    if (cell.use_count() != 1 || cell->borrowed) return cell->read();
    return cell->take();
}
struct Env {
    using Bindings = std::unordered_map<std::string, Ptr<Cell>>;
    Ptr<Bindings> bindings = std::make_shared<Bindings>();
    void writable() {
        if (bindings.use_count() != 1) bindings = std::make_shared<Bindings>(*bindings);
    }
    Ptr<Cell> cell(const std::string& name) const {
        auto found = bindings->find(name);
        if (found == bindings->end()) throw std::runtime_error("Missing generated binding: " + name);
        return found->second;
    }
    Value read(const std::string& name) const { return cell(name)->read(); }
    Value take(const std::string& name) {
        auto& binding = bindings->at(name);
        if (bindings.use_count() != 1 || binding.use_count() != 1 || binding->borrowed) return binding->read();
        return binding->take();
    }
    Env select(std::initializer_list<std::string> names) const {
        Env result;
        for (const auto& name : names) {
            auto found = bindings->find(name);
            if (found != bindings->end()) result.bindings->emplace(name,found->second);
        }
        return result;
    }
    void bind(const std::string& name, Value value, bool mutable_ = false) {
        writable();
        (*bindings)[name] = std::make_shared<Cell>(Cell{std::move(value), mutable_});
    }
    void bindCell(const std::string& name, const Ptr<Cell>& cell) {
        writable();(*bindings)[name]=cell;
    }
    void update(const std::string& name, Value value) { cell(name)->write(std::move(value)); }
    void erase(const std::string& name) { writable(); bindings->erase(name); }
};

// @runtime Borrow
struct Borrow { Ptr<Cell> owner; bool active = true, suspended = false; };
// @runtime BorrowScope
struct BorrowScope {
    Ptr<Borrow> ref, parent;
    explicit BorrowScope(const Ptr<Cell>& cell) {
        const auto value=cell->read();
        if (is<Ptr<Borrow>>(value)) {
            parent = as<Ptr<Borrow>>(value);
            if (!parent->active || parent->suspended) throw std::runtime_error("Invalid mutable reborrow");
            ref = std::make_shared<Borrow>(Borrow{parent->owner});
            parent->suspended = true;
        } else {
            if (cell->borrowed) throw std::runtime_error("Value is already borrowed");
            ref = std::make_shared<Borrow>(Borrow{cell});
            cell->borrowed = true;
        }
    }
    ~BorrowScope() {
        ref->active = false;
        if (parent) parent->suspended = false;
        else ref->owner->borrowed = false;
    }
    BorrowScope(const BorrowScope&) = delete;
};

// @runtime NativePairVector nativePairVector
template <typename Left, typename Right> struct NativePairVector final : VectorBase {
    using Row = std::tuple<Left,Right>;
    std::vector<Row> rows;
    mutable Values boxed;
    explicit NativePairVector(std::vector<Row> data) : rows(std::move(data)) {}
    std::size_t size() const override { return rows.size(); }
    bool native() const override { return true; }
    std::span<const Value> items() const override {
        if (boxed.empty() && !rows.empty()) {
            Values values;values.reserve(rows.size());
            for (const auto& row : rows) values.push_back(tuple({std::get<0>(row),std::get<1>(row)}));
            boxed=std::move(values);
        }
        return boxed;
    }
};
template <typename Left, typename Right>
Ptr<VectorBase> nativePairVector(std::vector<std::tuple<Left,Right>> rows) {
    return std::make_shared<NativePairVector<Left,Right>>(std::move(rows));
}
// @runtime zipNativeVectors
template <typename Left, typename Right, typename LeftSource, typename RightSource>
Ptr<VectorBase> zipNativeVectors(const LeftSource& left, const RightSource& right) {
    const NativeVectorReader<Left> a(left);
    const NativeVectorReader<Right> b(right);
    const auto size=std::min(a.size(),b.size());
    std::vector<std::tuple<Left,Right>> rows;rows.reserve(size);
    for (std::size_t i=0; i<size; ++i) rows.emplace_back(a[i],b[i]);
    return nativePairVector<Left,Right>(std::move(rows));
}
// @runtime unzipNativeVector
template <typename Left, typename Right, typename Source>
std::tuple<Ptr<VectorBase>,Ptr<VectorBase>> unzipNativeVector(const Source& source) {
    const auto* base=as<Ptr<VectorBase>>(source).get();
    std::vector<Left> left;left.reserve(base->size());
    std::vector<Right> right;right.reserve(base->size());
    if (const auto* pairs=dynamic_cast<const NativePairVector<Left,Right>*>(base)) {
        for (const auto& [a,b] : pairs->rows) { left.push_back(a);right.push_back(b); }
    } else {
        for (const auto& item : base->items()) {
            const auto& fields=as<Ptr<Tuple>>(item)->values;
            left.push_back(as<Left>(fields[0]));right.push_back(as<Right>(fields[1]));
        }
    }
    return {scalarVector<Left>(std::move(left)),scalarVector<Right>(std::move(right))};
}

// @runtime vectorPeek
template <typename T, typename Source>
Adt vectorPeek(const Source& source) {
    if constexpr (std::is_same_v<T,Value>) {
        const auto items=as<Ptr<VectorBase>>(source)->items();
        return typedOption(items.empty() ? std::optional<T>{} : std::optional<T>{items.back()});
    } else {
        const NativeVectorReader<T> items(source);
        return typedOption(items.size()==0 ? std::optional<T>{} : std::optional<T>{items[items.size()-1]});
    }
}
// @runtime vectorPop
template <typename T, typename Source>
std::tuple<Adt,Ptr<VectorBase>> vectorPop(Source source, bool reuse) {
    if (!as<Ptr<VectorBase>>(source)->size()) return {typedOption(std::optional<T>{}),as<Ptr<VectorBase>>(source)};
    auto data=vectorWritable<T>(std::move(source),reuse);
    T item=std::move(data->values.back());
    data->values.pop_back();
    return {typedOption(std::optional<T>{std::move(item)}),Ptr<VectorBase>(std::move(data))};
}

struct QueueNode {
    Value value; Ptr<QueueNode> next;
    QueueNode(Value value,Ptr<QueueNode> next):value(std::move(value)),next(std::move(next)){}
    ~QueueNode(){while(next&&next.use_count()==1){auto old=std::move(next);next=std::move(old->next);}}
};
struct Queue { Ptr<QueueNode> front,back; std::size_t size=0; };
inline Values queueItems(const Queue& queue){Values out,back;for(auto p=queue.front;p;p=p->next)out.push_back(p->value);for(auto p=queue.back;p;p=p->next)back.push_back(p->value);out.insert(out.end(),back.rbegin(),back.rend());return out;}
template <int Operation, bool KeepPeekState = false, typename Item = Value, typename Source>
inline auto queueOp(Source source,const Value& value){
    if constexpr (Operation==6) return static_cast<Int>(as<Ptr<Queue>>(source)->size);
    else if constexpr (Operation==7) return as<Ptr<Queue>>(source)->size==0;
    else {
        auto old=takeAs<Ptr<Queue>>(source);
        auto result=old.use_count()==1?std::move(old):std::make_shared<Queue>(*old);
        if constexpr (Operation<2) {
            auto& side=Operation==0?result->back:result->front;
            side=std::make_shared<QueueNode>(value,side);
            ++result->size;
            return result;
        } else {
            constexpr bool backwards=Operation==3||Operation==5;
            auto& side=backwards?result->back:result->front;
            auto& other=backwards?result->front:result->back;
            if(!side){for(auto p=other;p;p=p->next)side=std::make_shared<QueueNode>(p->value,side);other.reset();}
            std::optional<Item> item;
            if (side) {
                if constexpr (std::is_same_v<Item,Value>) item.emplace(side->value);
                else item.emplace(as<Item>(side->value));
                if constexpr (Operation<4) {side=side->next;--result->size;}
            }
            if constexpr (Operation<4 || KeepPeekState) return std::tuple{std::move(result),typedOption(std::move(item))};
            else return typedOption(std::move(item));
        }
    }
}
// @runtime truth

inline bool truth(bool value) { return value; }
// @runtime utf8
inline std::string utf8(char32_t code) {
    std::string result;
    if (code < 0x80) result += static_cast<char>(code);
    else if (code < 0x800) {
        result += static_cast<char>(0xc0 | (code >> 6));
        result += static_cast<char>(0x80 | (code & 63));
    } else if (code < 0x10000) {
        result += static_cast<char>(0xe0 | (code >> 12));
        result += static_cast<char>(0x80 | ((code >> 6) & 63));
        result += static_cast<char>(0x80 | (code & 63));
    } else {
        result += static_cast<char>(0xf0 | (code >> 18));
        result += static_cast<char>(0x80 | ((code >> 12) & 63));
        result += static_cast<char>(0x80 | ((code >> 6) & 63));
        result += static_cast<char>(0x80 | (code & 63));
    }
    return result;
}
// @runtime decode
inline std::optional<char32_t> decode(const std::string& text, std::size_t& pos) {
    auto lead = static_cast<Byte>(text[pos++]);
    if (lead < 0x80) return lead;
    int extra = lead >= 0xf0 ? 3 : lead >= 0xe0 ? 2 : 1;
    if (lead < 0xc2 || lead > 0xf4 || pos + extra > text.size()) return {};
    char32_t code = lead & (0x7f >> (extra + 1));
    for (int i = 0; i < extra; ++i) {
        auto part = static_cast<Byte>(text[pos++]);
        if ((part & 0xc0) != 0x80) return {};
        code = (code << 6) | (part & 63);
    }
    if (code < (extra == 1 ? 0x80 : extra == 2 ? 0x800 : 0x10000)
        || (code >= 0xd800 && code <= 0xdfff) || code > 0x10ffff) return {};
    return code;
}
// @runtime escape
inline std::string escape(const std::string& text, char quote) {
    std::string result(1, quote);
    for (unsigned char c : text) {
        if (c == quote || c == '\\') { result += '\\'; result += static_cast<char>(c); }
        else if (c == '\n') result += "\\n";
        else if (c == '\r') result += "\\r";
        else if (c == '\t') result += "\\t";
        else if (c == '\b') result += "\\b";
        else if (c == '\f') result += "\\f";
        else if (c < 32) {
            constexpr char digits[] = "0123456789abcdef";
            result += "\\u00";
            result += digits[c >> 4]; result += digits[c & 15];
        } else result += static_cast<char>(c);
    }
    return result + quote;
}

// @runtime join
inline std::string join(const std::vector<std::string>& parts, const std::string& sep) {
    std::string result;
    for (std::size_t i = 0; i < parts.size(); ++i) {
        if (i) result += sep;
        result += parts[i];
    }
    return result;
}
// @runtime shown
inline std::string shown(std::span<const Value> values, const std::string& sep) {
    std::vector<std::string> parts;
    for (const auto& value : values) parts.push_back(show(value));
    return join(parts, sep);
}
// @runtime floatText
inline std::string floatText(double n) {
    if (n == 0) return "0";
    if (!std::isfinite(n)) return std::isnan(n) ? "NaN" : n < 0 ? "-Infinity" : "Infinity";
    char buf[400];
    auto style = std::abs(n) >= 1e-6 && std::abs(n) < 1e21
        ? std::chars_format::fixed : std::chars_format::scientific;
    auto [end, error] = std::to_chars(buf, buf + sizeof(buf), n, style);
    std::string text(buf, end);
    auto exponent = text.find('e');
    if (exponent != std::string::npos && exponent + 3 < text.size() && text[exponent + 2] == '0')
        text.erase(exponent + 2, 1);
    return text;
}
// @runtime showScalar
template <typename T> std::string showScalar(const T& value) {
    if constexpr (std::is_same_v<T,bool>) return value ? "True" : "False";
    else if constexpr (std::is_same_v<T,double>) return floatText(value);
    else if constexpr (std::is_same_v<T,std::string>) return escape(value, '"');
    else return std::to_string(value);
}
// @runtime showChar
inline std::string showChar(Char value) {
    const auto code = value.code;
    if (code == 0) return "'\\0'";
    if ((code < 32 || code == 127) && code != 9 && code != 10 && code != 13) {
        char hex[8];
        auto end = std::to_chars(hex,hex+sizeof(hex),static_cast<unsigned>(code),16).ptr;
        return "'\\u{" + std::string(hex,end) + "}'";
    }
    return escape(utf8(code), '\'');
}
// @runtime showAdt
inline std::string showAdt(const Adt& a) {
    std::string name(a.info().display);
    if (!a.count()) return name;
    std::string text = "(" + name;
    for (std::size_t i=0; i<a.count(); ++i) text += " " + show(a.field(i));
    return text + ")";
}
// @runtime show
inline std::string show(const Value& v) {
    if (is<Int>(v)) return showScalar(as<Int>(v));
    if (is<Byte>(v)) return showScalar(as<Byte>(v));
    if (is<double>(v)) return showScalar(as<double>(v));
    if (is<bool>(v)) return showScalar(as<bool>(v));
// @if-runtime Char
    if (is<Char>(v)) return showChar(as<Char>(v));
// @end-runtime
    if (is<std::string>(v)) return showScalar(as<std::string>(v));
// @if-runtime VectorBase
    if (is<Ptr<VectorBase>>(v)) return "[" + shown(as<Ptr<VectorBase>>(v)->items(), ", ") + "]";
// @end-runtime
// @if-runtime Tuple
    if (is<Ptr<Tuple>>(v)) return "#(" + shown(as<Ptr<Tuple>>(v)->values, " ") + ")";
// @end-runtime
// @if-runtime Record
    if (is<Ptr<Record>>(v)) {
        std::vector<std::string> parts;
        for (const auto& [key, value] : as<Ptr<Record>>(v)->fields) parts.push_back(key + ": " + show(value));
        return parts.empty() ? "{:}" : "{" + join(parts, ", ") + "}";
    }
// @end-runtime
// @if-runtime HashSetBase
    if(is<Ptr<HashSetBase>>(v)){std::vector<std::string> parts;for(const auto& p:as<Ptr<HashSetBase>>(v)->keys())parts.push_back(show(p));return "set ["+join(parts,", ")+"]";}
// @end-runtime
// @if-runtime DictBase
    if (is<Ptr<DictBase>>(v)) {
        std::vector<std::string> parts;
        for (const auto& [key, value] : as<Ptr<DictBase>>(v)->entries()) parts.push_back(show(key) + ": " + show(value));
        return parts.empty() ? "[:]" : "[" + join(parts, ", ") + "]";
    }
// @end-runtime
// @if-runtime Adt
    if (is<Adt>(v)) return showAdt(as<Adt>(v));
// @end-runtime
// @if-runtime Range
    if (is<Range>(v)) {
        const auto& r = as<Range>(v);
        return "(" + std::to_string(r.first) + (r.inclusive ? " ..= " : " .. ") + std::to_string(r.last) + ")";
    }
// @end-runtime
// @if-runtime Queue
    if(is<Ptr<Queue>>(v))return "deque ["+shown(queueItems(*as<Ptr<Queue>>(v)),", ")+"]";
// @end-runtime
// @if-runtime Collection
    if (is<Ptr<Collection>>(v)) {
        const auto& c = *as<Ptr<Collection>>(v);
        return c.kind + " [" + shown(c.values, ", ") + "]";
    }
// @end-runtime
// @if-runtime Stream
    if (is<Ptr<Stream>>(v)) return "<stream>";
// @end-runtime
// @if-runtime Continuation
    if (is<Ptr<Continuation>>(v)) return "<continuation>";
// @end-runtime
// @if-runtime Borrow
    if (is<Ptr<Borrow>>(v)) return "<mutref>";
// @end-runtime
// @if-runtime Quote
    if (is<Ptr<Quote>>(v)) return "{ ... }";
// @end-runtime
    return "<value>";
}
// @runtime raw
inline std::string raw(const Value& v) { return is<std::string>(v) ? as<std::string>(v) : show(v); }

// @runtime order
template <typename T> int order(const T& a, const T& b) { return a < b ? -1 : b < a ? 1 : 0; }
// @runtime compare
inline int compare(const Value& a, const Value& b);
// @runtime compareAdtTag
inline int compareAdtTag(AdtTag left, AdtTag right) {
    return left.family != right.family ? order(left.family,right.family) : order(left.alternative,right.alternative);
}
// @runtime compareAdt
inline int compareAdt(const Adt& a, const Adt& b) {
    if (int rank=compareAdtTag(a.tag(),b.tag())) return rank;
    for (std::size_t i=0; i<a.count(); ++i)
        if (int rank=compare(a.field(i),b.field(i))) return rank;
    return 0;
}
// @runtime compareValues
inline int compareValues(std::span<const Value> a, std::span<const Value> b) {
    for (std::size_t i = 0; i < std::min(a.size(), b.size()); ++i)
        if (int result = compare(a[i], b[i])) return result;
    return order(a.size(), b.size());
}
// @runtime compare
inline int compare(const Value& a, const Value& b) {
    if (a.type() != b.type()) return a.type().before(b.type()) ? -1 : 1;
    if (is<Int>(a)) return order(as<Int>(a), as<Int>(b));
    if (is<double>(a)) return order(as<double>(a), as<double>(b));
    if (is<Byte>(a)) return order(as<Byte>(a), as<Byte>(b));
    if (is<bool>(a)) return order(truth(a), truth(b));
// @if-runtime Char
    if (is<Char>(a)) return order(as<Char>(a).code, as<Char>(b).code);
// @end-runtime
    if (is<std::string>(a)) return order(as<std::string>(a), as<std::string>(b));
// @if-runtime VectorBase
    if (is<Ptr<VectorBase>>(a)) return compareValues(as<Ptr<VectorBase>>(a)->items(), as<Ptr<VectorBase>>(b)->items());
// @end-runtime
// @if-runtime Tuple
    if (is<Ptr<Tuple>>(a)) return compareValues(as<Ptr<Tuple>>(a)->values, as<Ptr<Tuple>>(b)->values);
// @end-runtime
// @if-runtime Adt
    if (is<Adt>(a)) return compareAdt(as<Adt>(a),as<Adt>(b));
// @end-runtime
// @if-runtime Record
    if (is<Ptr<Record>>(a)) {
        const auto& x = as<Ptr<Record>>(a)->fields; const auto& y = as<Ptr<Record>>(b)->fields;
        auto left = x.begin(), right = y.begin();
        for (; left != x.end() && right != y.end(); ++left, ++right) {
            if (int result = order(left->first, right->first)) return result;
            if (int result = compare(left->second, right->second)) return result;
        }
        return order(x.size(), y.size());
    }
// @end-runtime
// @if-runtime Range
    if (is<Range>(a)) {
        const auto& x = as<Range>(a); const auto& y = as<Range>(b);
        if (int result = order(x.first, y.first)) return result;
        if (int result = order(x.last, y.last)) return result;
        return order(x.inclusive, y.inclusive);
    }
// @end-runtime
// @if-runtime DictBase
    if (is<Ptr<DictBase>>(a)) {
        auto x = as<Ptr<DictBase>>(a)->entries(), y = as<Ptr<DictBase>>(b)->entries();
        const auto less = [](const auto& p,const auto& q){return compare(p.first,q.first)<0;};
        std::stable_sort(x.begin(),x.end(),less);
        std::stable_sort(y.begin(),y.end(),less);
        for (std::size_t i=0; i<std::min(x.size(),y.size()); ++i) {
            if (int result=compare(x[i].first,y[i].first)) return result;
            if (int result=compare(x[i].second,y[i].second)) return result;
        }
        return order(x.size(),y.size());
    }
// @end-runtime
    throw std::runtime_error("Values cannot be ordered");
}
// @runtime equal
inline bool equal(const Value& a, const Value& b);
// @runtime equalAdt
inline bool equalAdt(const Adt& a, const Adt& b) {
    if (a.tag() != b.tag()) return false;
    for (std::size_t i=0; i<a.count(); ++i)
        if (!equal(a.field(i),b.field(i))) return false;
    return true;
}
// @runtime equalValues
inline bool equalValues(std::span<const Value> a, std::span<const Value> b) {
    return a.size() == b.size() && std::equal(a.begin(), a.end(), b.begin(), equal);
}
// @runtime equal
inline bool equal(const Value& a, const Value& b) {
    if (a.type() != b.type()) return false;
    if (is<double>(a)) return as<double>(a) == as<double>(b);
// @if-runtime Continuation
    if (is<Ptr<Continuation>>(a)) return as<Ptr<Continuation>>(a) == as<Ptr<Continuation>>(b);
// @end-runtime
// @if-runtime VectorBase
    if (is<Ptr<VectorBase>>(a)) return equalValues(as<Ptr<VectorBase>>(a)->items(), as<Ptr<VectorBase>>(b)->items());
// @end-runtime
// @if-runtime Tuple
    if (is<Ptr<Tuple>>(a)) return equalValues(as<Ptr<Tuple>>(a)->values, as<Ptr<Tuple>>(b)->values);
// @end-runtime
// @if-runtime Adt
    if (is<Adt>(a)) return equalAdt(as<Adt>(a),as<Adt>(b));
// @end-runtime
// @if-runtime Record
    if (is<Ptr<Record>>(a)) {
        const auto& x = as<Ptr<Record>>(a)->fields; const auto& y = as<Ptr<Record>>(b)->fields;
        return x.size() == y.size() && std::equal(x.begin(), x.end(), y.begin(), [](const auto& p, const auto& q) {
            return p.first == q.first && equal(p.second, q.second);
        });
    }
// @end-runtime
// @if-runtime HashSetBase
    if(is<Ptr<HashSetBase>>(a)){const auto& x=*as<Ptr<HashSetBase>>(a);const auto& y=*as<Ptr<HashSetBase>>(b);if(x.size()!=y.size())return false;for(const auto& p:x.keys())if(!y.contains(p))return false;return true;}
// @end-runtime
// @if-runtime DictBase
    if (is<Ptr<DictBase>>(a)) {
        const auto& x = *as<Ptr<DictBase>>(a); const auto& y = *as<Ptr<DictBase>>(b);
        if (x.size() != y.size()) return false;
        for (const auto& [key, value] : x.entries()) {
            auto other = y.get(key);
            if (!other || !equal(value,*other)) return false;
        }
        return true;
    }
// @end-runtime
    return compare(a,b) == 0;
}
// @runtime hashMix
inline std::uint32_t hashMix(std::uint32_t hash, std::uint32_t value) {
    return (hash ^ value) * 16777619u;
}
// @runtime hashText
inline std::uint32_t hashText(const std::string& text) {
    std::uint32_t hash = 2166136261u;
    for (unsigned char c : text) hash = hashMix(hash,c);
    return hash;
}
// @runtime hashValue
inline std::uint32_t hashValue(const Value& v);
// @runtime hashAdt
inline std::uint32_t hashAdt(const Adt& a) {
    auto hash = a.info().hashSeed;
    for (std::size_t i=0; i<a.count(); ++i) hash = hashMix(hash,hashValue(a.field(i)));
    return hash;
}
// @runtime hashValues
inline std::uint32_t hashValues(std::uint32_t hash, std::span<const Value> xs) {
    for (const auto& x : xs) hash = hashMix(hash,hashValue(x));
    return hash;
}
// @runtime hashNumericScalar
template <typename T> std::uint32_t hashNumericScalar(T value) {
    if constexpr (std::is_same_v<T,Int>) {
        const auto bits=static_cast<UInt>(value);
        return static_cast<std::uint32_t>(bits ^ (bits >> 32));
    } else if constexpr (std::is_same_v<T,Byte>) return value;
    else {
        auto bits=std::bit_cast<UInt>(value == 0 ? 0.0 : value);
        std::uint32_t hash=2166136261u;
        for (int i=0; i<8; ++i, bits >>= 8) hash=hashMix(hash,static_cast<Byte>(bits));
        return hash;
    }
}
// @runtime hashValue
inline std::uint32_t hashValue(const Value& v) {
    if (is<Int>(v)) return hashNumericScalar(as<Int>(v));
    if (is<Byte>(v)) return hashNumericScalar(as<Byte>(v));
// @if-runtime Char
    if (is<Char>(v)) return as<Char>(v).code;
// @end-runtime
    if (is<bool>(v)) return truth(v) ? 1u : 0u;
    if (is<std::string>(v)) return hashText(as<std::string>(v));
    if (is<double>(v)) return hashNumericScalar(as<double>(v));
// @if-runtime VectorBase
    if (is<Ptr<VectorBase>>(v)) return hashValues(hashText("vector"),as<Ptr<VectorBase>>(v)->items());
// @end-runtime
// @if-runtime Tuple
    if (is<Ptr<Tuple>>(v)) return hashValues(hashText("tuple"),as<Ptr<Tuple>>(v)->values);
// @end-runtime
// @if-runtime Record
    if (is<Ptr<Record>>(v)) {
        auto hash = hashText("record");
        for (const auto& [k,x] : as<Ptr<Record>>(v)->fields) hash = hashMix(hashMix(hash,hashText(k)),hashValue(x));
        return hash;
    }
// @end-runtime
// @if-runtime Range
    if (is<Range>(v)) {
        const auto& r = as<Range>(v);
        return hashValues(hashText("range"), Values{r.first,r.last,r.inclusive});
    }
// @end-runtime
// @if-runtime Adt
    if (is<Adt>(v)) return hashAdt(as<Adt>(v));
// @end-runtime
// @if-runtime HashSetBase
    if(is<Ptr<HashSetBase>>(v)){const auto& set=*as<Ptr<HashSetBase>>(v);std::uint32_t sum=0;for(const auto& p:set.keys())sum+=set.hash(p);return hashMix(hashMix(hashText("set"),set.size()),sum);}
// @end-runtime
// @if-runtime DictBase
    if (is<Ptr<DictBase>>(v)) {
        const auto& d = *as<Ptr<DictBase>>(v);
        std::uint32_t sum = 0;
        for (const auto& [k,x] : d.entries()) sum += hashMix(static_cast<std::uint32_t>(d.hash(k)),hashValue(x));
        return hashMix(hashMix(hashText("dict"),static_cast<std::uint32_t>(d.size())),sum);
    }
// @end-runtime
    throw std::runtime_error("Missing hash instance");
}

// @runtime Pattern PatternKind
enum class PatternKind { word, literal, constructor, vector, push, tuple, record, alternative, concatenate, concatenateString };
struct Pattern {
    PatternKind kind;
    std::string name;
    Values literals;
    std::vector<Pattern> fields;
    std::vector<std::string> keys;
    bool mutable_ = false;
    AdtTag constructor{};
    int rest = -1;
};
// @runtime match
inline bool match(const Pattern&, const Value&, Env&);
// @runtime matches
inline bool matches(const std::vector<Pattern>& patterns, std::span<const Value> values, Env& env) {
    if (patterns.size() != values.size()) return false;
    for (std::size_t i = 0; i < patterns.size(); ++i)
        if (!match(patterns[i], values[i], env)) return false;
    return true;
}
// @runtime match
inline bool match(const Pattern& p, const Value& value, Env& env) {
    switch (p.kind) {
    case PatternKind::alternative:
        for (const auto& choice : p.fields) {
            Env trial;
            if (!match(choice,value,trial)) continue;
            env.writable();
            for (const auto& [name,binding] : *trial.bindings) (*env.bindings)[name] = binding;
            return true;
        }
        return false;
    case PatternKind::word:
        if (p.name != "_") env.bind(p.name, value, p.mutable_);
        return true;
    case PatternKind::literal:
        if (is<Int>(p.literals[0]) && is<Byte>(value)) return as<Int>(p.literals[0]) == as<Byte>(value);
        if (is<Int>(p.literals[0]) && is<double>(value)) return static_cast<double>(as<Int>(p.literals[0])) == as<double>(value);
        return equal(p.literals[0], value);
    case PatternKind::concatenateString: {
        if (!is<std::string>(value)) return false;
        const std::string_view text = as<std::string>(value);
        const auto& prefix = as<std::string>(p.literals[0]);
        const auto& suffix = as<std::string>(p.literals[1]);
        const auto fixed = prefix.size() + suffix.size();
        if (text.size() < fixed || !text.starts_with(prefix) || !text.ends_with(suffix)) return false;
        if (p.fields.empty()) return text.size() == fixed;
        return match(p.fields[0], std::string(text.substr(prefix.size(), text.size()-fixed)), env);
    }
// @if-runtime Adt
    case PatternKind::constructor: {
        const auto* adt = adtPointer(value);
        if (!adt || adt->tag() != p.constructor || adt->count() != p.fields.size()) return false;
        for (std::size_t i=0; i<p.fields.size(); ++i)
            if (!match(p.fields[i],adt->field(i),env)) return false;
        return true;
    }
// @end-runtime
// @if-runtime VectorBase
    case PatternKind::vector:
        return is<Ptr<VectorBase>>(value) && matches(p.fields, as<Ptr<VectorBase>>(value)->items(), env);
    case PatternKind::concatenate: {
        if (!is<Ptr<VectorBase>>(value)) return false;
        const auto xs = as<Ptr<VectorBase>>(value)->items();
        std::size_t fixed=0, offset=0;
        for (std::size_t i=0;i<p.fields.size();++i)
            if (static_cast<int>(i)!=p.rest) fixed+=p.fields[i].fields.size();
        if (xs.size()<fixed || (p.rest<0 && xs.size()!=fixed)) return false;
        for (std::size_t i=0;i<p.fields.size();++i) {
            auto count=static_cast<int>(i)==p.rest ? xs.size()-fixed : p.fields[i].fields.size();
            if (!match(p.fields[i],vectorView(value,offset,count),env)) return false;
            offset+=count;
        }
        return true;
    }
    case PatternKind::push: {
        if (!is<Ptr<VectorBase>>(value)) return false;
        const auto& xs = as<Ptr<VectorBase>>(value)->items();
        return !xs.empty() && matches(p.fields, Values{vector(Values(xs.begin(), xs.end()-1)), xs.back()}, env);
    }
// @end-runtime
// @if-runtime Tuple
    case PatternKind::tuple:
        return is<Ptr<Tuple>>(value) && matches(p.fields, as<Ptr<Tuple>>(value)->values, env);
// @end-runtime
// @if-runtime Record
    case PatternKind::record: {
        if (!is<Ptr<Record>>(value)) return false;
        const auto& fields = as<Ptr<Record>>(value)->fields;
        for (std::size_t i = 0; i < p.fields.size(); ++i) {
            auto it = fields.find(p.keys[i]);
            if (it == fields.end() || !match(p.fields[i], it->second, env)) return false;
        }
        return true;
    }
// @end-runtime
    default:
        return false;
    }
}

// @runtime Event Route
struct Event {
    std::string name, at;
    Values arguments;
    bool never = false;
    const void* target = nullptr;
    unsigned skip = 0;
};
using Route = std::function<Values(Event)>;
// @runtime Handler
using Handler = std::function<Values(const Values&, Ptr<Continuation>)>;
// @runtime Machine !perform !exception
// @if-runtime WorklistRoute
struct WorklistRoute;
// @end-runtime
struct Machine {
    Env& globals;
    Route route;
// @if-runtime WorklistRoute
    WorklistRoute* worklist = nullptr;
// @end-runtime
    Values perform(Event event);
    [[noreturn]] void exception(const std::string& message, const std::string& at);
};
// @runtime perform
inline Values Machine::perform(Event event) {
    if (route) return route(std::move(event));
    if (event.target || event.skip) fail(event.at, "Unhandled routed effect: " + event.name);
    if (event.name == "put") {
        std::cout << raw(event.arguments.at(0)) << '\n';
        return {};
    }
    if (event.name == "read") {
#ifndef NYANCO_BATCH_IO
        std::cout.flush();
#endif
        std::string line;
        std::getline(std::cin, line);
        if (!line.empty() && line.back() == '\r') line.pop_back();
        return {line};
    }
    if (event.name == "exn") fail(event.at, "Exception: " + raw(event.arguments.at(0)));
    fail(event.at, "Unhandled effect: " + event.name);
}
// @runtime exception
inline void Machine::exception(const std::string& message, const std::string& at) {
    perform(Event{"exn", at, {message}, true});
    fail(at, "Non-returning effect returned");
}
// @runtime RoutedLabel
struct RoutedLabel {
    Ptr<Label> label;
    const void* previous = nullptr;
    explicit RoutedLabel(Ptr<Label> value) : label(std::move(value)) {
        if (label) { previous = label->target; label->target = this; }
    }
    ~RoutedLabel() { if (label) label->target = previous; }
    RoutedLabel(const RoutedLabel&) = delete;
};
// @runtime labelTarget
inline const void* labelTarget(const Value& value, const std::string& at) {
    const auto& label = as<Ptr<Label>>(value);
    if (!label->target) fail(at, "Effect label is outside its handler lifetime");
    return label->target;
}
// @runtime accepts
inline bool accepts(Event& event, const RoutedLabel& scope) {
    if (event.target) return event.target == &scope;
    if (event.skip) { --event.skip; return false; }
    return true;
}
// @runtime Abort
struct Abort { const void* owner; std::optional<Values> result; };
// @runtime loop
template <typename Body>
std::optional<Values> loop(Machine& parent, Body body, Ptr<Label> label = {}, bool cancelOnly = false) {
    const RoutedLabel scope(std::move(label));
    Machine child{parent.globals, [&](Event event) -> Values {
        if ((event.name == "cancel" || (!cancelOnly && event.name == "break")) && accepts(event, scope))
            throw Abort{&scope, event.name == "break" ? std::optional<Values>(std::move(event.arguments)) : std::nullopt};
        return parent.perform(std::move(event));
    }};
// @if-runtime WorklistRoute
    child.worklist = parent.worklist;
// @end-runtime
    try { body(child); }
    catch (Abort& stop) {
        if (stop.owner != &scope) throw;
        return std::move(stop.result);
    }
    return std::nullopt;
}
// @runtime resume
inline Values resume(const Value& value, Values arguments, const std::string& at) {
    const auto& k = as<Ptr<Continuation>>(value);
    if (k->never) fail(at, "Cannot resume a non-returning effect");
    if (!k->active || k->used) fail(at, "Continuation is expired or already resumed");
    k->used = true;
    return k->resume(std::move(arguments));
}
// @runtime resumeTail
inline bool resumeTail(const Value& value, const Ptr<Continuation>& current, Values arguments, const std::string& at) {
    const auto& k = as<Ptr<Continuation>>(value);
    if (k.get() != current.get()) return false;
    if (k->never) fail(at, "Cannot resume a non-returning effect");
    if (!k->active || k->used) fail(at, "Continuation is expired or already resumed");
    k->used = true;
    k->tailReply.emplace(std::move(arguments));
    return true;
}

// Suspending a worker retains the C++ call frames across a user effect.
// The driver and worker alternate at each operation. Only arguments and
// results cross the rendezvous, and abandoning a continuation joins the worker.
// @runtime EffectWorker Abandoned
struct Abandoned {};
class EffectWorker {
    std::mutex mutex;
    std::condition_variable changed;
    std::optional<Event> pending;
    std::optional<Values> reply;
    bool done = false, cancelled = false;
    Values result;
    std::exception_ptr failure;
    std::thread worker;
    Values request(Event event) {
        std::unique_lock lock(mutex);
        if (cancelled) throw Abandoned{};
        pending = std::move(event);
        changed.notify_all();
        changed.wait(lock, [&]{ return cancelled || reply.has_value(); });
        if (cancelled) throw Abandoned{};
        Values values = std::move(*reply);
        reply.reset();
        return values;
    }
public:
    template <typename Body>
    EffectWorker(Env& globals, Body body) : worker([this, &globals, body = std::move(body)] {
        Machine child{globals, [this](Event event){ return request(std::move(event)); }};
        try { result = body(child); }
        catch (...) { failure = std::current_exception(); }
        std::lock_guard lock(mutex);
        done = true;
        changed.notify_all();
    }) {}
    ~EffectWorker() {
        {
            std::lock_guard lock(mutex);
            cancelled = true;
            changed.notify_all();
        }
        worker.join();
    }
    std::optional<Event> next() {
        std::unique_lock lock(mutex);
        changed.wait(lock, [&]{ return done || pending.has_value(); });
        if (done) {
            if (failure) std::rethrow_exception(failure);
            return std::nullopt;
        }
        auto event = std::move(pending);
        pending.reset();
        return event;
    }
    Values finish() { return std::move(result); }
    void send(Values values) {
        std::lock_guard lock(mutex);
        reply = std::move(values);
        changed.notify_all();
    }
};
// @runtime handle
template <typename Body>
Values handle(Machine& parent, Body body, const std::map<std::string, Handler>& handlers, Ptr<Label> label = {}) {
    const RoutedLabel scope(std::move(label));
    EffectWorker worker(parent.globals, std::move(body));
    std::function<Values()> drive = [&]() -> Values {
        while (auto event = worker.next()) {
            auto handler = handlers.find(event->name);
            if (handler == handlers.end() || !accepts(*event, scope)) {
                worker.send(parent.perform(std::move(*event)));
                continue;
            }
            auto k = std::make_shared<Continuation>();
            k->never = event->never;
            k->resume = [&](Values values) {
                worker.send(std::move(values));
                return drive();
            };
            struct Expire { Ptr<Continuation> k; ~Expire() { k->active = false; k->resume = {}; } } expire{k};
            auto answer = handler->second(event->arguments, k);
// @if-runtime resumeTail
            if (k->tailReply) {
                // Unwind this handler before servicing another operation.
                worker.send(std::move(*k->tailReply));
                continue;
            }
// @end-runtime
            return answer;
        }
        return worker.finish();
    };
    return drive();
}
// @runtime mask
template <typename Body>
Values mask(Machine& parent, Body body, const std::string& name, const void* target) {
    Machine child{parent.globals, [&](Event event) {
        if (event.name == name && !event.target) {
            if (target) event.target = target;
            else ++event.skip;
        }
        return parent.perform(std::move(event));
    }};
    return body(child);
}
// @runtime labelBlock
template <typename Body>
Values labelBlock(Machine& parent, Body body) {
    auto label = std::make_shared<Label>();
    const RoutedLabel scope(label);
    Machine child{parent.globals, [&](Event event) {
        if (event.target == &scope) event.target = nullptr;
        return parent.perform(std::move(event));
    }};
// @if-runtime WorklistRoute
    child.worklist = parent.worklist;
// @end-runtime
    return body(child, label);
}

// @runtime readBorrow
inline Value readBorrow(const Value& value, const std::string& at) {
    const auto& ref = as<Ptr<Borrow>>(value);
    if (!ref->active || ref->suspended) fail(at, "Invalid mutable reference");
    return ref->owner->read();
}
// @runtime writeBorrow
inline void writeBorrow(const Value& value, Value replacement, const std::string& at) {
    const auto& ref = as<Ptr<Borrow>>(value);
    if (!ref->active || ref->suspended) fail(at, "Invalid mutable reference");
    ref->owner->write(std::move(replacement));
}
// @runtime vectorOffset
inline Int vectorOffset(Int index, std::size_t size) {
    return index < 0 ? index + static_cast<Int>(size) : index;
}
// @runtime vectorPosition
inline Int vectorPosition(Int index, std::size_t size) {
    const auto pos = vectorOffset(index,size);
    return pos < 0 || static_cast<UInt>(pos) >= size ? -1 : pos;
}
// @runtime stringFind
inline std::optional<Byte> stringFind(const std::string& text, Int index) {
    if (static_cast<UInt>(index) >= text.size()) return {};
    return static_cast<Byte>(text[static_cast<std::size_t>(index)]);
}
// @runtime stringAt
inline Byte stringAt(Machine& m, const std::string& text, Int index, const std::string& at) {
    if (auto item=stringFind(text,index)) return *item;
    m.exception("Index out of bounds: " + showScalar(index),at);
}
// @runtime stringSetAt
inline std::string stringSetAt(Machine& m,std::string text,Int index,Byte replacement,const std::string& at) {
    if (static_cast<UInt>(index) >= text.size()) m.exception("Index out of bounds: " + showScalar(index),at);
    text[static_cast<std::size_t>(index)] = static_cast<char>(replacement);
    for (std::size_t i=0; i<text.size();)
        if (!decode(text,i)) m.exception("String update produces invalid UTF-8",at);
    return text;
}
// @runtime findIndex
inline std::optional<Value> findIndex(const Value& value, const Value& index) {
    if (is<Ptr<DictBase>>(value)) {
        const auto& dict = *as<Ptr<DictBase>>(value);
        if (auto value = dict.get(index)) return *value;
        return {};
    }
    Int pos = as<Int>(index);
    if (is<std::string>(value)) {
        if (auto item=stringFind(as<std::string>(value),pos)) return *item;
    } else {
        const auto& xs = as<Ptr<VectorBase>>(value)->items();
        pos = vectorOffset(pos,xs.size());
        if (static_cast<UInt>(pos) < xs.size()) return xs[static_cast<std::size_t>(pos)];
    }
    return {};
}
// @runtime dictFind
template <typename T>
std::optional<T> dictFind(const Ptr<DictBase>& dict, const Value& key) {
    auto found=dict->get(key);
    if constexpr (std::is_same_v<T,Value>) return found;
    else if (found) return takeAs<T>(*found);
    else return {};
}
// @runtime dictAt
template <typename T>
T dictAt(Machine& m,const Ptr<DictBase>& dict,const Value& key,const std::string& at) {
    if (auto found=dictFind<T>(dict,key)) return std::move(*found);
    m.exception("Dictionary key not found: " + show(key),at);
}
// @runtime dictSetAt
inline Ptr<DictBase> dictSetAt(Ptr<DictBase> dict,const Value& key,const Value& replacement,bool reuse) {
    if (!reuse || dict.use_count() != 1) dict = dict->clone();
    dict->set(key,replacement);
    return dict;
}
// @runtime atIndex
inline Value atIndex(Machine& m, const Value& value, const Value& index, const std::string& at) {
    if (auto item = findIndex(value, index)) return *item;
    m.exception((is<Ptr<DictBase>>(value) ? "Dictionary key not found: " : "Index out of bounds: ") + show(index), at);
}
// @runtime vectorSliceBounds
inline std::optional<std::pair<std::size_t,std::size_t>> vectorSliceBounds(std::size_t length, Int first, Int last, bool from) {
    const auto size = static_cast<Int>(length);
    const auto start = vectorOffset(first,length), end = from ? size : vectorOffset(last,length);
    if (start < 0 || start > size || end < 0 || end > size) return {};
    return std::pair{static_cast<std::size_t>(start),end<start ? 0 : static_cast<std::size_t>(end-start)};
}
// @runtime sliceVector
template <typename Source>
inline Ptr<VectorBase> sliceVector(Machine& m, const Source& value, Int first, Int last,
                         bool from, const std::string& at) {
    const auto bounds = vectorSliceBounds(as<Ptr<VectorBase>>(value)->size(),first,last,from);
    if (!bounds) m.exception("Slice bounds out of range",at);
    return vectorView(value,bounds->first,bounds->second);
}
// @runtime vectorSetAt
template <typename Source, typename Item>
inline Ptr<VectorBase> vectorSetAt(Machine& m,Source source,Int index,const Item& replacement,const std::string& at,bool reuse) {
    const auto pos=vectorPosition(index,as<Ptr<VectorBase>>(source)->size());
    if (pos<0) m.exception("Index out of bounds: " + showScalar(index),at);
    auto xs=vectorWritable<Value>(std::move(source),reuse);
    xs->values[static_cast<std::size_t>(pos)] = replacement;
    return xs;
}
// @runtime setIndex
inline Value setIndex(Machine& m, Value value, const Value& index, const Value& replacement,
                      const std::string& at, bool reuse = false) {
    if (is<Ptr<DictBase>>(value)) {
        return dictSetAt(takeAs<Ptr<DictBase>>(value),index,replacement,reuse);
    }
    if (is<std::string>(value)) {
        return stringSetAt(m,takeAs<std::string>(value),as<Int>(index),as<Byte>(replacement),at);
    }
    return vectorSetAt(m,std::move(value),as<Int>(index),replacement,at,reuse);
}
// @runtime append
template <bool Concat, typename Source, typename Item>
inline Ptr<VectorBase> append(Source value, const Item& item, bool reuse) {
    if constexpr (Concat) if(as<Ptr<VectorBase>>(value)==as<Ptr<VectorBase>>(item))reuse=false;
    auto xs = vectorWritable<Value>(std::move(value),reuse);
    if constexpr (Concat) {
        const auto& ys = as<Ptr<VectorBase>>(item)->items();
        xs->values.insert(xs->values.end(), ys.begin(), ys.end());
    } else xs->values.push_back(item);
    return Ptr<VectorBase>(std::move(xs));
}
// @runtime field
template <typename Source> inline const Value& field(const Source& value, const std::string& key) { return as<Ptr<Record>>(value)->fields.at(key); }
// @runtime tupleAt
template <typename Source> inline const Value& tupleAt(const Source& value, std::size_t i) { return as<Ptr<Tuple>>(value)->values.at(i); }
// @runtime UpdatePart
struct UpdatePart {
    std::string kind, field;
    Value index;
    bool optional;
    std::function<Value(Machine&, const Value&, const Value&)> get;
    std::function<Value(Machine&, const Value&, const Value&, const Value&)> set;
};
// @runtime pathRead
inline Value pathRead(Machine& m, const Value& value, const UpdatePart& part, const std::string& at) {
    if (part.kind == "field") return field(value, part.field);
    if (part.kind == "tuple-index") return tupleAt(value, static_cast<std::size_t>(as<Int>(part.index)));
    return part.get ? part.get(m,value,part.index) : atIndex(m,value,part.index,at);
}
// @runtime pathFind
inline std::optional<Value> pathFind(Machine& m, const Value& value, const UpdatePart& part) {
    if (!part.get) return findIndex(value,part.index);
    auto values = handle(m,[&](Machine& child) -> Values {return {part.get(child,value,part.index)};},
        {{"exn",[](const Values&, Ptr<Continuation>) -> Values {return {};}}});
    if (values.empty()) return std::nullopt;
    return std::move(values[0]);
}
// @runtime updatePath
template <typename Replace>
Value updatePath(Machine& m, Value value, const std::vector<UpdatePart>& path, std::size_t i,
                 const Replace& replace, bool defaulting, bool updating, bool reuse, const std::string& at) {
    if (i == path.size()) return replace(std::move(value));
    const auto& part = path[i];
    const bool leaf = i + 1 == path.size();
    Value old;
    if (part.optional || (defaulting && leaf)) {
        auto found = pathFind(m, value, part);
        if (part.optional && !found) return value;
        if (defaulting && leaf && found) return value;
        if (found) old = *found;
    } else if (!leaf || updating) old = pathRead(m, value, part, at);
    if (part.kind == "field") {
        auto result = std::move(std::any_cast<Ptr<Record>&>(value));
        if (!reuse || result.use_count() != 1) result = std::make_shared<Record>(*result);
        old.reset();
        if (!leaf || updating) old = std::move(result->fields.at(part.field));
        result->fields[part.field] = updatePath(m,std::move(old),path,i+1,replace,defaulting,updating,reuse,at);
        return result;
    }
    if (part.kind == "tuple-index") {
        auto result = std::move(std::any_cast<Ptr<Tuple>&>(value));
        if (!reuse || result.use_count() != 1) result = std::make_shared<Tuple>(*result);
        auto& slot = result->values.at(static_cast<std::size_t>(as<Int>(part.index)));
        old.reset();
        if (!leaf || updating) old = std::move(slot);
        slot = updatePath(m,std::move(old),path,i+1,replace,defaulting,updating,reuse,at);
        return result;
    }
    if ((!leaf || updating) && !part.get && !part.set && is<Ptr<VectorBase>>(value)) {
        atIndex(m,value,part.index,at);
        auto result = vectorWritable<Value>(std::move(value),reuse);
        auto& slot = result->values.at(static_cast<std::size_t>(vectorOffset(as<Int>(part.index),result->size())));
        old.reset();
        if (!leaf || updating || part.optional) old = std::move(slot);
        slot = updatePath(m,std::move(old),path,i+1,replace,defaulting,updating,reuse,at);
        return Ptr<VectorBase>(std::move(result));
    }
    auto changed = updatePath(m,std::move(old),path,i+1,replace,defaulting,updating,reuse,at);
    return part.set ? part.set(m,value,part.index,changed) : setIndex(m,std::move(value),part.index,changed,at,reuse);
}

// @runtime VectorIndex updateVectorAt
struct VectorIndex { Int index; bool optional; std::string_view at{}; };
template <typename T>
Value writeVectorUpdate(Value source, std::span<const VectorIndex> indices, const Value& replacement, bool reuse) {
    const auto pos = static_cast<std::size_t>(vectorOffset(indices.front().index,as<Ptr<VectorBase>>(source)->size()));
    if (indices.size() == 1) {
        auto result = vectorWritable<T>(std::move(source), reuse);
        if constexpr (std::is_same_v<T,Value>) result->values[pos] = replacement;
        else result->values[pos] = as<T>(replacement);
        return Ptr<VectorBase>(std::move(result));
    }
    auto result = vectorWritable<Value>(std::move(source), reuse);
    auto& slot = result->values[pos];
    slot = writeVectorUpdate<T>(std::move(slot), indices.subspan(1), replacement, reuse);
    return Ptr<VectorBase>(std::move(result));
}
template <typename T, typename Replace>
Value updateVectorAt(Machine& m, Value source, std::initializer_list<VectorIndex> indices,
                     const Replace& replace, bool reuse, const Ptr<Cell>& cell, const std::string& at) {
    const auto* container = as<Ptr<VectorBase>>(source).get();
    Value old;
    for (const auto& part : indices) {
        const auto index = vectorOffset(part.index,container->size());
        if (index < 0 || static_cast<UInt>(index) >= container->size()) {
            if (part.optional) return source;
            m.exception("Index out of bounds: " + show(Value{part.index}), part.at.empty()?at:std::string(part.at));
        }
        const auto pos = static_cast<std::size_t>(index);
        if (&part != indices.end() - 1) {
            container = as<Ptr<VectorBase>>(container->items()[pos]).get();
        } else if (const auto* typed = dynamic_cast<const VectorStorage<T>*>(container)) {
            old = typed->scalars()[pos];
        } else {
            old = container->items()[pos];
        }
    }
    auto replacement = replace(std::move(old));
    if (reuse && cell) {
        bool same;
        {
            const auto current = cell->read();
            same = is<Ptr<VectorBase>>(current) &&
                as<Ptr<VectorBase>>(current).get() == as<Ptr<VectorBase>>(source).get();
        }
        // The callback may have replaced the cell. Only consume the cell
        // when it still contains the snapshot used to compute the result.
        if (same) {
            source.reset();
            source = cell->take();
        }
    }
    return writeVectorUpdate<T>(std::move(source), {indices.begin(), indices.size()}, replacement, reuse);
}

// @runtime bits
inline Int bits(UInt n) { return std::bit_cast<Int>(n); }
// @runtime Arithmetic
enum class Arithmetic { add, subtract, multiply, divide, remainder, power };
// @runtime intOp
template <Arithmetic Op>
inline Int intOp(Int x, Int y, const std::string& at) {
    auto a = static_cast<UInt>(x), b = static_cast<UInt>(y);
    if constexpr (Op == Arithmetic::add) return bits(a+b);
    else if constexpr (Op == Arithmetic::subtract) return bits(a-b);
    else if constexpr (Op == Arithmetic::multiply) return bits(a*b);
    else if constexpr (Op == Arithmetic::divide || Op == Arithmetic::remainder) {
        if (!y) fail(at, "Division by zero");
        if (x == std::numeric_limits<Int>::min() && y == -1) return Op == Arithmetic::divide ? x : 0;
        return Op == Arithmetic::divide ? x/y : x%y;
    } else {
        if (y < 0) fail(at, "Integer exponent must be nonnegative");
        UInt result = 1;
        for (; b; b >>= 1, a *= a) if (b & 1) result *= a;
        return bits(result);
    }
}
// @runtime byteOp
template <Arithmetic Op>
inline Byte byteOp(Byte x, Byte y, const std::string& at) {
    Int result;
    if constexpr (Op == Arithmetic::power) {
        result = 1;
        for (Int i = 0; i < y; ++i) {
            result *= x;
            if (result > 255) fail(at, "byte overflow");
        }
    } else result = intOp<Op>(x,y,at);
    if (result < 0 || result > 255) fail(at, "byte overflow");
    return static_cast<Byte>(result);
}
// @runtime floatOp
template <Arithmetic Op>
inline double floatOp(double x, double y, const std::string& at) {
    double result;
    if constexpr (Op == Arithmetic::add) result = x+y;
    else if constexpr (Op == Arithmetic::subtract) result = x-y;
    else if constexpr (Op == Arithmetic::multiply) result = x*y;
    else if constexpr (Op == Arithmetic::divide || Op == Arithmetic::remainder) {
        if (y == 0) fail(at, "Division by zero");
        result = Op == Arithmetic::divide ? x/y : std::fmod(x,y);
    } else result = std::pow(x,y);
    if (!std::isfinite(result)) fail(at, "Non-finite numeric result");
    return result;
}
// @runtime bitPosition
inline unsigned bitPosition(Machine& m, Int index, unsigned width, const std::string& at) {
    if (index < 0 || static_cast<UInt>(index) >= width)
        m.exception("Bit position or shift count is outside 0.." + std::to_string(width-1),at);
    return static_cast<unsigned>(index);
}
// @runtime BitOp
enum class BitOp { bitAnd, bitOr, bitXor, shiftLeft, shiftRight, testBit, setBit, clearBit };
// @runtime bitOp
template <BitOp Op>
inline Value bitOp(Machine& m, const Value& left, const Value& right, const std::string& at) {
    if constexpr (Op == BitOp::bitAnd || Op == BitOp::bitOr || Op == BitOp::bitXor) {
        if (is<bool>(left)) {
            if constexpr (Op == BitOp::bitAnd) return truth(left) && truth(right);
            else if constexpr (Op == BitOp::bitOr) return truth(left) || truth(right);
            else return truth(left) != truth(right);
        }
    }
    bool byte = is<Byte>(left);
    UInt x = byte ? as<Byte>(left) : static_cast<UInt>(as<Int>(left));
    UInt y = is<Byte>(right) ? as<Byte>(right) : static_cast<UInt>(as<Int>(right));
    auto width = byte ? 8u : 64u;
    UInt result = 0;
    if constexpr (Op == BitOp::bitAnd) result = x & y;
    else if constexpr (Op == BitOp::bitOr) result = x | y;
    else if constexpr (Op == BitOp::bitXor) result = x ^ y;
    else {
        const unsigned pos = bitPosition(m,bits(y),width,at);
        const UInt mask = UInt{1} << pos;
        if constexpr (Op == BitOp::testBit) return (x & mask) != 0;
        else if constexpr (Op == BitOp::setBit) result = x | mask;
        else if constexpr (Op == BitOp::clearBit) result = x & ~mask;
        else if constexpr (Op == BitOp::shiftLeft) result = x << pos;
        else result = byte ? x >> pos : static_cast<UInt>(bits(x) >> pos);
    }
    if (byte) return static_cast<Byte>(result);
    return bits(result);
}
// @runtime walkRange
template <typename Visitor> inline void walkRange(const Range& r, const Visitor& visit) {
    const Int step = r.first > r.last ? -1 : 1;
    for (Int n = r.first; n != r.last; n += step) visit(n);
    if (r.inclusive) visit(r.last);
}

    if (is<Int>(a)) return intOp<Op>(as<Int>(a),as<Int>(b),at);
    if (is<Byte>(a)) return byteOp<Op>(as<Byte>(a),as<Byte>(b),at);
    return floatOp<Op>(as<double>(a),as<double>(b),at);
}

// @runtime visitChars

void visitChars(const std::string& text, const std::string& at, const Visitor& visit) {
    for (std::size_t pos = 0; pos < text.size();) {
        auto code = decode(text,pos);
        if (!code) fail(at, "Invalid UTF-8");
        visit(Char{*code});
    }
}

    visitChars(text,at,visit);
}

    auto data = vectorWritable<T>(std::move(source),reuse);

    return Ptr<VectorBase>(std::move(data));
}

// @runtime cast
template <typename Target, typename Source>
inline Target cast(Machine& m, const Source& value, const std::string& at) {
    if constexpr (std::is_same_v<Target,std::string>) {
        return utf8(value.code);
    } else if constexpr (std::is_same_v<Source,std::string>) {
        static_assert(std::is_same_v<Target,Char>);
        std::size_t pos = 0;
        auto code = value.empty() ? std::optional<char32_t>{} : decode(value,pos);
        if (!code || pos != value.size()) m.exception("toChar requires one character",at);
        return Char{*code};
    } else if constexpr (std::is_same_v<Target,double>) {
        if constexpr (std::is_same_v<Source,Char>) return static_cast<double>(value.code);
        else return static_cast<double>(value);
    } else {
        constexpr const char* target = std::is_same_v<Target,Byte> ? "byte" :
            std::is_same_v<Target,Char> ? "char" : "int";
        auto invalid = [&] { m.exception(std::string("Invalid numeric cast to ")+target,at); };
        Int n;
        if constexpr (std::is_same_v<Source,double>) {
            if (!std::isfinite(value) || value < -0x1p63 || value >= 0x1p63) invalid();
            n = static_cast<Int>(value);
        } else if constexpr (std::is_same_v<Source,Char>) n = value.code;
        else n = value;
        if constexpr (std::is_same_v<Target,Byte>) {
            if (n < 0 || n > 255) invalid();
            return static_cast<Byte>(n);
        } else if constexpr (std::is_same_v<Target,Char>) {
            if (n < 0 || n > 0x10ffff || (n >= 0xd800 && n <= 0xdfff)) invalid();
            return Char{static_cast<char32_t>(n)};
        } else {
            static_assert(std::is_same_v<Target,Int>);
            return n;
        }
    }
}
// @runtime parseNumber parseNumberText ParsedNumber
template <typename Number, bool Strict>
using ParsedNumber = std::conditional_t<Strict,Number,Adt>;
template <typename Number, bool Strict>
inline ParsedNumber<Number,Strict> parseNumberText(Machine& m, std::string_view text, const std::string& at) {
    auto first = text.find_first_not_of(" \t\r\n\f\v");
    auto last = text.find_last_not_of(" \t\r\n\f\v");
    auto start = text.data() + (first == std::string::npos ? text.size() : first);
    auto end = text.data() + (last == std::string::npos ? text.size() : last+1);
    if (start < end && *start == '+') ++start;
    std::optional<Number> result;
    constexpr bool integer = std::is_same_v<Number,Int>;
    if constexpr (integer) {
        Int n; auto parsed = std::from_chars(start,end,n);
        if (parsed.ec == std::errc{} && parsed.ptr == end) result = n;
    } else {
        double n; auto parsed = std::from_chars(start,end,n,std::chars_format::general);
        if (parsed.ec == std::errc{} && parsed.ptr == end && std::isfinite(n)) result = n;
    }
    if constexpr (Strict) {
        if (!result) m.exception(std::string(integer ? "Invalid integer: " : "Invalid number: ") + show(Value{std::string(text)}),at);
        return *result;
    } else return typedOption(result);
}
template <typename Number, bool Strict>
inline ParsedNumber<Number,Strict> parseNumber(Machine& m, const Value& value, const std::string& at) {
    return parseNumberText<Number,Strict>(m,as<std::string>(value),at);
}
// @runtime walk
// @if-runtime VectorBase
inline void walk(Machine&, const Ptr<VectorBase>& source, const Visit& visit) {
    for (const auto& value : source->items()) visit(value);
}
// @end-runtime
// @if-runtime Stream
inline void walk(Machine& m, const Ptr<Stream>& source, const Visit& visit) {
    source->walk(m,visit);
}
// @end-runtime
// @if-runtime Range
inline void walk(Machine&, const Range& source, const Visit& visit) {
    walkRange(source,visit);
}
// @end-runtime
inline void walk(Machine& m, const Value& value, const Visit& visit) {
// @if-runtime VectorBase
    if (is<Ptr<VectorBase>>(value)) { walk(m,as<Ptr<VectorBase>>(value),visit); return; }
// @end-runtime
// @if-runtime Range
    if (is<Range>(value)) { walk(m,as<Range>(value),visit); return; }
// @end-runtime
// @if-runtime Stream
    if (is<Ptr<Stream>>(value)) { walk(m,as<Ptr<Stream>>(value),visit); return; }
// @end-runtime
// @if-runtime HashSetBase
    if(is<Ptr<HashSetBase>>(value)){for(const auto& p:as<Ptr<HashSetBase>>(value)->keys())visit(p);return;}
// @end-runtime
// @if-runtime DictBase
    if (is<Ptr<DictBase>>(value)) {
        for (const auto& [k,v] : as<Ptr<DictBase>>(value)->entries()) visit(tuple({k,v}));
        return;
    }
// @end-runtime
// @if-runtime Queue
    if(is<Ptr<Queue>>(value)){for(const auto& item:queueItems(*as<Ptr<Queue>>(value)))visit(item);return;}
// @end-runtime
// @if-runtime Collection
    if (is<Ptr<Collection>>(value)) {
        for (const auto& v : as<Ptr<Collection>>(value)->values) visit(v);
        return;
    }
// @end-runtime
    throw std::runtime_error("Value is not iterable");
}
// @runtime stream
inline Ptr<Stream> stream(std::function<void(Machine&, const Visit&)> body) {
    return std::make_shared<Stream>(std::move(body));
}
// @runtime TypedVisit TypedWalk
template <typename T> using TypedVisit = std::function<void(T)>;
template <typename T> using TypedWalk = std::function<void(Machine&, const TypedVisit<T>&)>;
// @runtime TypedStream typedStream
// Untyped consumers use the boxed Stream ABI. Typed consumers visit scan
// directly, so boxing is confined to a boundary between the two ABIs.
template <typename T> struct TypedStream final : Stream {
    TypedWalk<T> scan;
    explicit TypedStream(TypedWalk<T> body) : Stream([this](Machine& m, const Visit& visit) {
        scan(m,[&](T typedItem) { visit(std::move(typedItem)); });
    }), scan(std::move(body)) {}
};
template <typename T> Ptr<Stream> typedStream(TypedWalk<T> body) {
    return std::make_shared<TypedStream<T>>(std::move(body));
}
// @runtime walkTyped
// @if-runtime VectorBase
template <typename T> void walkTyped(Machine&, const Ptr<VectorBase>& value, const TypedVisit<T>& visit) {
    if constexpr (std::is_same_v<T,Int> || std::is_same_v<T,Byte> || std::is_same_v<T,double>) {
        const NativeVectorReader<T> values(value);
        for (std::size_t i=0; i<values.size(); ++i) visit(values[i]);
    } else {
        for (const auto& item : value->items()) visit(as<T>(item));
    }
}
// @end-runtime
// @if-runtime Range
template <typename T> void walkTyped(Machine&, const Range& source, const TypedVisit<T>& visit) {
    walkRange(source,visit);
}
// @end-runtime
template <typename T> void walkTyped(Machine& m, const Ptr<Stream>& source, const TypedVisit<T>& visit) {
    if (const auto* typed = dynamic_cast<const TypedStream<T>*>(source.get())) {
        typed->scan(m,visit);
        return;
    }
    walk(m,source,[&](Value item) { visit(takeAs<T>(item)); });
}
template <typename T> void walkTyped(Machine& m, const Value& value, const TypedVisit<T>& visit) {
    if (is<Ptr<Stream>>(value)) { walkTyped<T>(m,as<Ptr<Stream>>(value),visit); return; }
// @if-runtime VectorBase
    if (is<Ptr<VectorBase>>(value)) {
        walkTyped<T>(m,as<Ptr<VectorBase>>(value),visit);
        return;
    }
// @end-runtime
// @if-runtime Range
    if constexpr (std::is_same_v<T,Int>)
        if (is<Range>(value)) { walkTyped<T>(m,as<Range>(value),visit); return; }
// @end-runtime
    walk(m,value,[&](Value item) { visit(takeAs<T>(item)); });
}

// @runtime fromYield
inline Ptr<Stream> fromYield(Ptr<Label> label, std::function<void(Machine&)> body) {
    return stream([label, body = std::move(body)](Machine& parent, const Visit& visit) {
        const RoutedLabel scope(label);
        Machine child{parent.globals,[&](Event event) -> Values {
            if (event.name == "yield" && accepts(event,scope)) {
                visit(event.arguments.at(0));
                return {};
            }
            return parent.perform(std::move(event));
        }};
        body(child);
    });
}
// @runtime StopIteration untilStopped
struct StopIteration { const void* owner; };
template <typename Body> void untilStopped(Body body) {
    int owner = 0;
    try { body([&]{ throw StopIteration{&owner}; }); }
    catch (StopIteration& stop) { if (stop.owner != &owner) throw; }
}

    if (count < (positive ? 1 : 0)) m.exception(op + (positive ? ": count must be positive" : ": count must be nonnegative"),at);
    return count;
}
// @runtime MapItem
using MapItem = std::function<Value(Machine&, Value)>;
// @runtime Walk
using Walk = std::function<void(Machine&, const Visit&)>;
// @runtime WalkItem
using WalkItem = std::function<void(Machine&, Value, const Visit&)>;
// @runtime Transform
enum class Transform { map, filter, takeWhile, skipWhile, take, skip, flatMap, flatten };
// @runtime mapNativePairs
template <typename Left, typename Right, typename Out = Value, typename Source, typename Map>
Ptr<Stream> mapNativePairs(Source source, Map fn, Ptr<Label> label) {
    auto body = [source=std::move(source),fn=std::move(fn),label](Machine& parent,const TypedVisit<Out>& visit) {
        loop(parent,[&](Machine& m){
            const auto* base=as<Ptr<VectorBase>>(source).get();
            if (const auto* pairs=dynamic_cast<const NativePairVector<Left,Right>*>(base)) {
                for (const auto& row : pairs->rows) visit(fn(m,std::get<0>(row),std::get<1>(row)));
            } else {
                for (const auto& item : base->items()) {
                    const auto& fields=as<Ptr<Tuple>>(item)->values;
                    visit(fn(m,as<Left>(fields[0]),as<Right>(fields[1])));
                }
            }
        },label);
    };
    if constexpr (std::is_same_v<Out,Value>) return stream(std::move(body));
    else return typedStream<Out>(std::move(body));
}
// @runtime transformTyped
template <typename PredicateResult> bool streamPredicate(const PredicateResult& result) {
    if constexpr (std::is_same_v<PredicateResult,bool>) return result;
    else return truth(result);
}
template <Transform Mode, typename In, typename Out, typename Map, typename Nested = decltype(nullptr)>
Ptr<Stream> transformTyped(TypedWalk<In> source, Int count, Map fn, Ptr<Label> label, bool cancelOnly, Nested nested = nullptr) {
    auto body = [source=std::move(source),count,fn=std::move(fn),label,cancelOnly,nested=std::move(nested)]
        (Machine& parent,const TypedVisit<Out>& visit) {
        if constexpr (Mode == Transform::take) if (!count) return;
        Int remaining = count;
        bool skipping = true;
        untilStopped([&](auto stop) {
            loop(parent,[&](Machine& m) {
                source(m,[&](In typedItem) {
                    if constexpr (Mode == Transform::map) visit(fn(m,std::move(typedItem)));
                    else if constexpr (Mode == Transform::take) {
                        visit(std::move(typedItem));
                        if (!--remaining) stop();
                    } else if constexpr (Mode == Transform::skip) {
                        if (remaining) --remaining; else visit(std::move(typedItem));
                    } else if constexpr (Mode == Transform::flatten) nested(m,std::move(typedItem),visit);
                    else if constexpr (Mode == Transform::flatMap) nested(m,fn(m,std::move(typedItem)),visit);
                    else if constexpr (Mode == Transform::skipWhile) {
                        if (skipping && streamPredicate(fn(m,typedItem))) return;
                        skipping = false;
                        visit(std::move(typedItem));
                    } else if constexpr (Mode == Transform::filter) {
                        if (streamPredicate(fn(m,typedItem))) visit(std::move(typedItem));
                    } else if constexpr (Mode == Transform::takeWhile) {
                        if (!streamPredicate(fn(m,typedItem))) stop();
                        visit(std::move(typedItem));
                    }
                });
            },label,cancelOnly);
        });
    };
    if constexpr (std::is_same_v<Out,Value>) return stream(std::move(body));
    else return typedStream<Out>(std::move(body));
}
// @runtime transform
template <Transform Mode>
inline Ptr<Stream> transform(Walk source, Int count, MapItem fn, Ptr<Label> label, bool cancelOnly, WalkItem nested) {
    return transformTyped<Mode,Value,Value>(std::move(source),count,std::move(fn),label,cancelOnly,std::move(nested));
}

// @runtime SplitInput splitParts
struct SplitInput { std::string text, separator; };
template <typename VisitPart>
inline void splitParts(const SplitInput& input, const VisitPart& visit) {
    const auto& text = input.text;
    const auto& sep = input.separator;
    if (sep.empty()) {
        for (std::size_t pos=0;pos<text.size();) {
            auto start=pos;
            if (!decode(text,pos)) throw std::runtime_error("Invalid UTF-8");
            visit(std::string_view(text).substr(start,pos-start));
        }
        return;
    }
    for (std::size_t pos=0;;) {
        auto end=text.find(sep,pos);
        visit(std::string_view(text).substr(pos,end==std::string::npos?end:end-pos));
        if (end==std::string::npos) return;
        pos=end+sep.size();
    }
}
// @runtime split
template <bool Typed, typename Source, typename Separator>
inline Ptr<Stream> split(const Source& source, const Separator& separator) {
    using Item = std::conditional_t<Typed,std::string,Value>;
    auto body = [input = SplitInput{as<std::string>(source),as<std::string>(separator)}]
        (Machine&, const TypedVisit<Item>& visit) {
        splitParts(input,[&](std::string_view part) { visit(std::string(part)); });
    };
    if constexpr (Typed) return typedStream<Item>(std::move(body));
    else return stream(std::move(body));
}
// @runtime Compare
using Compare = std::function<bool(Machine&,const Value&,const Value&)>;
// @runtime equalScalarVectors
template <typename T, typename Left, typename Right>
bool equalScalarVectors(Machine&, const Left& a, const Right& b) {
    const auto& x = as<Ptr<VectorBase>>(a);
    const auto& y = as<Ptr<VectorBase>>(b);
    if constexpr (!std::is_same_v<T,double>) if (x == y) return true;
    const auto* left = dynamic_cast<const ScalarVector<T>*>(x.get());
    const auto* right = dynamic_cast<const ScalarVector<T>*>(y.get());
    if (!left || !right) return equalValues(x->items(),y->items());
    const auto xs = left->scalars(), ys = right->scalars();
    return std::equal(xs.begin(),xs.end(),ys.begin(),ys.end());
}
// @runtime compareScalarVectors
template <typename T, typename Left, typename Right> int compareScalarVectors(const Left& a, const Right& b) {
    const NativeVectorReader<T> xs(a),ys(b);
    for (std::size_t i=0; i<std::min(xs.size(),ys.size()); ++i)
        if (const int rank=order(xs[i],ys[i])) return rank;
    return order(xs.size(),ys.size());
}
// @runtime hashScalarVector
template <typename T, typename Source> std::uint32_t hashScalarVector(const Source& value) {
    const NativeVectorReader<T> xs(value);
    auto hash=hashText("vector");
    for (std::size_t i=0; i<xs.size(); ++i) hash=hashMix(hash,hashNumericScalar(xs[i]));
    return hash;
}
// @runtime groupOrderedCounts
template <typename Key, typename Equal>
Value groupOrderedCounts(Machine& m, Value sorted, Equal eq) {
    const auto items = as<Ptr<VectorBase>>(sorted)->items();
    std::vector<std::tuple<Key,Int>> result;
    if (!items.empty()) {
        std::size_t first = 0;
        const auto emit = [&](std::size_t end) {
            result.emplace_back(as<Key>(items[first]),static_cast<Int>(end-first));
        };
        for (std::size_t i=1; i<items.size(); ++i) {
            if (eq(m,items[i],items[first])) continue;
            emit(i);
            first=i;
        }
        emit(items.size());
    }
    return nativePairVector<Key,Int>(std::move(result));
}
// @runtime Partition partition
enum class Partition { windows, chunks, chunkBy };
template <Partition Mode, bool CountGroups = false>
inline Value partition(Walk source, Int count, MapItem key, Compare eq, Ptr<Label> label) {
    return stream([=](Machine& parent,const Visit& visit) {
        loop(parent,[&](Machine& m){
            Values pending;
            Value last;
            Int groupSize = 0;
            const auto emit = [&] {
                if constexpr (CountGroups) visit(groupSize);
                else if constexpr (Mode == Partition::chunkBy) visit(stream([saved=pending](Machine&,const Visit& visit){
                    for (const auto& value : saved) visit(value);
                }));
                else visit(vector(pending));
            };
            source(m,[&](Value item){
                if constexpr (Mode == Partition::chunkBy) {
                    auto k = key ? key(m,item) : CountGroups ? std::move(item) : item;
                    if (groupSize && !(eq ? eq(m,last,k) : equal(last,k))) { emit(); groupSize=0; pending.clear(); }
                    last = std::move(k);
                    ++groupSize;
                    if constexpr (!CountGroups) pending.push_back(std::move(item));
                } else {
                    pending.push_back(std::move(item));
                    if (pending.size() == static_cast<UInt>(count)) {
                        emit();
                        if constexpr (Mode == Partition::windows) pending.erase(pending.begin());
                        else pending.clear();
                    }
                }
            });
            if constexpr (Mode != Partition::windows) if (groupSize || !pending.empty()) emit();
        },label);
    });
}
// @runtime Cursor
class Cursor {
    Walk walk;
    Ptr<EffectWorker> worker;
    bool waiting = false;
public:
    explicit Cursor(Walk source) : walk(std::move(source)) {}
    std::optional<Value> next(Machine& m) {
        if (!worker) worker = std::make_shared<EffectWorker>(m.globals,[this](Machine& child) -> Values {
            walk(child,[&](Value item){ child.perform(Event{"@zip-item","<zip>",{std::move(item)}}); });
            return {};
        });
        else if (waiting) worker->send({});
        waiting = false;
        while (auto event = worker->next()) {
            if (event->name == "@zip-item") { waiting = true; return std::move(event->arguments[0]); }
            worker->send(m.perform(std::move(*event)));
        }
        return {};
    }
};
// @runtime zip
inline Ptr<Stream> zip(Walk left, Walk right, Ptr<Label> label) {
    return stream([=](Machine& parent,const Visit& visit) {
        loop(parent,[&](Machine& m){
            Cursor a(left),b(right);
            while (auto x = a.next(m)) {
                auto y = b.next(m);
                if (!y) break;
                visit(tuple({std::move(*x),std::move(*y)}));
            }
        },label,true);
    });
}
// @runtime zipVectors
template <typename Left, typename Right>
inline Ptr<Stream> zipVectors(Left left,Right right,Ptr<Label> label){
    return stream([left=std::move(left),right=std::move(right),label](Machine& parent,const Visit& visit){
        loop(parent,[&](Machine&){
            const auto& xs=as<Ptr<VectorBase>>(left)->items();const auto& ys=as<Ptr<VectorBase>>(right)->items();
            for(std::size_t i=0;i<std::min(xs.size(),ys.size());i++)visit(tuple({xs[i],ys[i]}));
        },label);
    });
}
// @runtime Combinatorics combinatorics
enum class Combinatorics { permutations, combinations, combinationsWithReplacement, powerset, cartesianProduct };
template <Combinatorics Mode>
inline Ptr<Stream> combinatorics(Walk left, Walk right, Int count) {
    return stream([=](Machine& m,const Visit& visit){
        Values xs,ys;
        loop(m,[&](Machine& child){ left(child,[&](Value v){xs.push_back(std::move(v));}); });
        if constexpr (Mode == Combinatorics::cartesianProduct) {
            loop(m,[&](Machine& child){ right(child,[&](Value v){ys.push_back(std::move(v));}); });
            for (const auto& a : xs) for (const auto& b : ys) visit(tuple({a,b}));
            return;
        }
        const auto emit = [&](UInt k) {
            if (!k) { visit(vector({})); return; }
            if (xs.empty() || (k > xs.size() && Mode != Combinatorics::combinationsWithReplacement)) return;
            std::vector<std::size_t> selected, next{0};
            while (!next.empty()) {
                if (selected.size() == k) {
                    Values choice;
                    for (auto i : selected) choice.push_back(xs[i]);
                    visit(vector(std::move(choice)));
                    next.pop_back(); selected.pop_back();
                    continue;
                }
                auto i = next.back()++;
                if (i == xs.size()) {
                    next.pop_back(); if (!selected.empty()) selected.pop_back();
                    continue;
                }
                if (i > xs.size()) { next.pop_back(); if (!selected.empty()) selected.pop_back(); continue; }
                if constexpr (Mode == Combinatorics::permutations)
                    if (std::find(selected.begin(),selected.end(),i) != selected.end()) continue;
                selected.push_back(i);
                next.push_back(Mode == Combinatorics::permutations ? 0 : Mode == Combinatorics::combinationsWithReplacement ? i : i+1);
            }
        };
        if constexpr (Mode == Combinatorics::powerset) for (std::size_t k=0; k<=xs.size(); ++k) emit(k);
        else emit(static_cast<UInt>(count));
    });
}
// @runtime CollectionOp collectionOp
enum class CollectionOp { hpush, hpop, hpeek, hlength, hisEmpty };
template <CollectionOp Op, typename Item = Value, typename Source, typename Less>
inline auto collectionOp(Source source, const Value& item, const Less& less, bool reuse) {
    if constexpr (Op == CollectionOp::hlength) return static_cast<Int>(as<Ptr<Collection>>(source)->values.size());
    else if constexpr (Op == CollectionOp::hisEmpty) return as<Ptr<Collection>>(source)->values.empty();
    else if constexpr (Op == CollectionOp::hpeek) {
        const auto& xs=as<Ptr<Collection>>(source)->values;
        std::optional<Item> found;
        if (!xs.empty()) {
            if constexpr (std::is_same_v<Item,Value>) found.emplace(xs.front());
            else found.emplace(as<Item>(xs.front()));
        }
        return typedOption(std::move(found));
    }
    else {
        auto old=takeAs<Ptr<Collection>>(source);
        auto result=reuse && old.use_count()==1 ? std::move(old) : std::make_shared<Collection>(*old);
        auto& xs=result->values;
        if constexpr (Op == CollectionOp::hpush) {
            xs.push_back(item);
            std::push_heap(xs.begin(),xs.end(),less);
            return result;
        } else {
            std::optional<Item> popped;
            if (!xs.empty()) {
                std::pop_heap(xs.begin(),xs.end(),less);
                if constexpr (std::is_same_v<Item,Value>) popped.emplace(std::move(xs.back()));
                else popped.emplace(takeAs<Item>(xs.back()));
                xs.pop_back();
            }
            return std::tuple{std::move(result),typedOption(std::move(popped))};
        }
    }
}
// @runtime WorklistRoute WorklistOperation nativeWorklistTarget
enum class WorklistOperation { Push, Pop, Peek };
struct WorklistRoute {
    const RoutedLabel* scope = nullptr;
    virtual ~WorklistRoute() = default;
    virtual Value runWorklist(WorklistOperation operation, const Value& item) = 0;
};
inline WorklistRoute* nativeWorklistTarget(Machine& m, const void* target) {
    auto* state = m.worklist;
    return state && (!target || target == state->scope) ? state : nullptr;
}
// @runtime NativeHeapWorklist
template <typename Less> struct NativeHeapWorklist final : WorklistRoute {
    Ptr<Collection> pending;
    Less less;
    NativeHeapWorklist(Ptr<Collection> initial, Less compare)
        : pending(std::move(initial)), less(std::move(compare)) {}
    Value runWorklist(WorklistOperation operation, const Value& item) override {
        if (operation == WorklistOperation::Peek)
            return collectionOp<CollectionOp::hpeek>(pending, {}, less, true);
        if (operation == WorklistOperation::Push) {
            pending = collectionOp<CollectionOp::hpush>(std::move(pending),item,less,true);
            return {};
        }
        auto [heap, option] = collectionOp<CollectionOp::hpop>(std::move(pending),{},less,true);
        pending = std::move(heap);
        return std::move(option);
    }
};
// @runtime NativeDequeWorklist
struct NativeDequeWorklist final : WorklistRoute {
    Ptr<Queue> pending;
    explicit NativeDequeWorklist(Ptr<Queue> initial) : pending(std::move(initial)) {}
    Value runWorklist(WorklistOperation operation, const Value& item) override {
        if (operation == WorklistOperation::Push) {
            pending = queueOp<0>(std::move(pending),item);
            return {};
        }
        // Retain the normalized front so repeated peeks do not reverse the back again.
        auto [queue, option] = operation == WorklistOperation::Pop
            ? queueOp<2>(std::move(pending), {}) : queueOp<4,true>(std::move(pending), {});
        pending = std::move(queue);
        return std::move(option);
    }
};
// @runtime withNativeWorklist
template <typename Body>
Values withNativeWorklist(Machine& parent, WorklistRoute& state, Body body, Ptr<Label> label) {
    const RoutedLabel scope(std::move(label));
    state.scope = &scope;
    Machine child{parent.globals, [&](Event event) -> Values {
        std::optional<WorklistOperation> operation;
        if (event.name == "__worklistPush") operation = WorklistOperation::Push;
        else if (event.name == "__worklistPop") operation = WorklistOperation::Pop;
        else if (event.name == "__worklistPeek") operation = WorklistOperation::Peek;
        if (!operation || !accepts(event, scope)) return parent.perform(std::move(event));
        if (*operation == WorklistOperation::Push) {
            state.runWorklist(*operation, event.arguments.at(0));
            return {};
        }
        return {state.runWorklist(*operation, {})};
    }};
    child.worklist = &state;
    return body(child);
}
// @runtime Consumption
struct Consumption { Value value; std::optional<Values> broken; };
// @runtime Consume consume
enum class Consume { join, nth, reverse, sort, sortBy, maxi, mini, maxiBy, miniBy, all, any, dict, counts, groupBy };
template <Consume Mode>
inline Consumption consume(Machine& parent, Walk source, const Value& arg,
                           MapItem key, Compare less, Ptr<DictBase> dict, Ptr<Label> label) {
    Values collected;
    std::vector<std::pair<Value,Value>> sorted;
    std::vector<std::string> joined;
    Value result;
    std::optional<Value> best, bestKey;
    bool flag = Mode == Consume::all;
    Int remaining = Mode == Consume::nth ? as<Int>(arg) : 0;
    std::optional<Values> broken;
    if (remaining >= 0) untilStopped([&](auto stop){
        broken = loop(parent,[&](Machine& m){
            source(m,[&](Value item){
                if constexpr (Mode == Consume::join) joined.push_back(as<std::string>(item));
                else if constexpr (Mode == Consume::reverse) collected.push_back(item);
                else if constexpr (Mode == Consume::sort || Mode == Consume::sortBy) sorted.emplace_back(item,Mode == Consume::sortBy ? key(m,item) : item);
                else if constexpr (Mode == Consume::nth) { if (remaining-- == 0) { best = item; stop(); } }
                else if constexpr (Mode == Consume::all || Mode == Consume::any) {
                    flag = truth(key(m,item));
                    if (flag == (Mode == Consume::any)) stop();
                } else if constexpr (Mode == Consume::dict) dict->set(tupleAt(item,0),tupleAt(item,1));
                else if constexpr (Mode == Consume::counts) {
                    auto previous = dict->get(item);
                    dict->set(item, Int{previous ? as<Int>(*previous)+1 : 1});
                } else if constexpr (Mode == Consume::groupBy) {
                    auto k = key(m,item);
                    auto previous = dict->get(k);
                    if (previous) std::static_pointer_cast<Vector>(as<Ptr<VectorBase>>(*previous))->values.push_back(item);
                    else dict->set(k,vector({item}));
                } else {
                    auto k = Mode == Consume::maxiBy || Mode == Consume::miniBy ? key(m,item) : item;
                    constexpr bool maximum = Mode == Consume::maxi || Mode == Consume::maxiBy;
                    if (!bestKey || (maximum ? !less(m,k,*bestKey) : less(m,k,*bestKey))) {
                        best = item; bestKey = std::move(k);
                    }
                }
            });
        },label);
    });
    if constexpr (Mode == Consume::dict || Mode == Consume::counts || Mode == Consume::groupBy) result = dict;
    else if constexpr (Mode == Consume::all || Mode == Consume::any) result = flag;
    else if constexpr (Mode == Consume::join) result = join(joined,as<std::string>(arg));
    else if constexpr (Mode == Consume::sort || Mode == Consume::sortBy) {
        if (!broken) broken = loop(parent,[&](Machine& m) {
            std::stable_sort(sorted.begin(),sorted.end(),[&](const auto& a,const auto& b){return less(m,a.second,b.second);});
        },label);
        for (auto& pair : sorted) collected.push_back(std::move(pair.first));
        result = vector(std::move(collected));
    } else if constexpr (Mode == Consume::reverse) {
        std::reverse(collected.begin(),collected.end());
        result = vector(std::move(collected));
    } else result = option(best);
    return {std::move(result),std::move(broken)};
}
// @runtime sortScalarVectors
template <typename T, typename Item>
Consumption sortScalarVectors(Machine& parent, TypedWalk<Item> source, Ptr<Label> label) {
    struct Row { Ptr<ScalarVector<T>> value; UInt prefix; };
    std::vector<Row> rows;
    auto broken = loop(parent,[&](Machine& m){
        source(m,[&](Item typedItem){
            auto data = scalarData<T>(typedItem);
            UInt prefix = 0;
            if constexpr (std::is_same_v<T,Byte>) {
                const auto bytes = data->scalars();
                for (std::size_t i=0; i<std::min(bytes.size(),sizeof(UInt)); ++i)
                    prefix |= static_cast<UInt>(bytes[i]) << ((sizeof(UInt)-1-i)*8);
            }
            rows.push_back({std::move(data),prefix});
        });
    },label);
    if (!broken) std::stable_sort(rows.begin(),rows.end(),[](const auto& a,const auto& b){
        if constexpr (std::is_same_v<T,Byte>) if (a.prefix != b.prefix) return a.prefix < b.prefix;
        auto x=a.value->scalars(),y=b.value->scalars();
        if constexpr (std::is_same_v<T,Byte>) {
            // Equal zero-padded prefixes still distinguish a shorter key.
            if (std::min(x.size(),y.size()) <= sizeof(UInt)) return x.size() < y.size();
            x=x.subspan(sizeof(UInt)); y=y.subspan(sizeof(UInt));
        }
        return std::lexicographical_compare(x.begin(),x.end(),y.begin(),y.end());
    });
    Values result;result.reserve(rows.size());
    for (auto& row : rows) result.emplace_back(Ptr<VectorBase>(std::move(row.value)));
    return {vector(std::move(result)),std::move(broken)};
}
// @runtime testStream
template <bool All, typename T, typename Predicate>
std::pair<bool,std::optional<Values>> testStream(Machine& parent, TypedWalk<T> source, Predicate fn, Ptr<Label> label) {
    bool result = All;
    std::optional<Values> broken;
    untilStopped([&](auto stop) {
        broken = loop(parent,[&](Machine& m) {
            source(m,[&](T typedItem) {
                result = fn(m,std::move(typedItem));
                if (result != All) stop();
            });
        },label);
    });
    return {result,std::move(broken)};
}
