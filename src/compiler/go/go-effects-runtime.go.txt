// Suspended Go calls retain locals on their goroutine call stacks. These
// messages carry only operation arguments/results, not an evaluation stack.
type EffectTarget struct{live bool}
type EffectLabel struct{current *EffectTarget}
func labelTarget(v any,loc string) *EffectTarget {h,ok:=v.(*EffectLabel);if !ok||h.current==nil||!h.current.live{fail(loc,"Effect label is outside its handler lifetime")};return h.current}
type EffectHandler func([]any,*Continuation) []any
type effectReply struct{values []any}
type effectEvent struct{name,loc string;args []any;reply chan effectReply;result []any;failure any;done bool;never bool;target *EffectTarget;skip int}
type EffectScope struct{events chan effectEvent;cancel chan struct{}}
type effectCancelled struct{}
type Continuation struct{used,active,never bool;resume func([]any) []any}
func resumeContinuation(value any,args []any,loc string) []any {
 k,ok:=value.(*Continuation);if !ok{fail(loc,"resume requires a continuation")}
 if k.never{fail(loc,"Cannot resume a non-returning effect")}
 if k.used||!k.active{fail(loc,"Continuation is expired or already resumed")}
 k.used=true;return k.resume(args)
}
func(m *Machine) perform(name string,args []any,loc string,never bool) []any {return m.performRoute(name,args,loc,never,nil,0)}
func(m *Machine) performRoute(name string,args []any,loc string,never bool,target *EffectTarget,skip int) []any {
 if m.effects==nil{if target!=nil||skip>0{fail(loc,"Unhandled routed effect: "+name)};if name=="exn"{fail(loc,"Exception: "+args[0].(string))};if name=="put"{printValue(args[0]);return nil};if name=="read"{return []any{readLine(loc)}};fail(loc,"Unhandled effect: "+name)}
 reply:=make(chan effectReply)
 event:=effectEvent{name:name,loc:loc,args:args,reply:reply,never:never,target:target,skip:skip}
 select{case m.effects.events<-event:case <-m.effects.cancel:panic(effectCancelled{})}
 select{case r:=<-reply:return r.values;case <-m.effects.cancel:panic(effectCancelled{})}
}
func(m *Machine) handle(body func(*Machine) []any,handlers map[string]EffectHandler,loc string,labels ...*EffectLabel) []any {var label *EffectLabel;if len(labels)>0{label=labels[0]};return m.handleWith(body,handlers,loc,label,"",nil)}
func(m *Machine) mask(body func(*Machine) []any,name string,target *EffectTarget,loc string) []any {return m.handleWith(body,nil,loc,nil,name,target)}
func(m *Machine) loop(body func(*Machine),loc string,label *EffectLabel,cancelOnly ...bool) []any {
 handlers:=map[string]EffectHandler{"cancel":func(args []any,k *Continuation) []any{return nil}}
 if len(cancelOnly)==0||!cancelOnly[0]{handlers["break"]=func(args []any,k *Continuation) []any{return append([]any{},args...)}}
 return m.handle(func(m *Machine) []any{body(m);return nil},handlers,loc,label)
}
func(m *Machine) handleWith(body func(*Machine) []any,handlers map[string]EffectHandler,loc string,label *EffectLabel,maskName string,maskTarget *EffectTarget,boundary ...bool) []any {
 target:=&EffectTarget{true};defer func(){target.live=false}()
 if label!=nil{previous:=label.current;label.current=target;defer func(){label.current=previous}()}

 scope:=&EffectScope{make(chan effectEvent),make(chan struct{})};finished:=make(chan struct{})
 child:=*m;child.effects=scope
 go func(){
  defer close(finished)
  event:=effectEvent{done:true}
  defer func(){
   if err:=recover();err!=nil{event.failure=err}
   select{case scope.events<-event:case <-scope.cancel:}
  }()
  event.result=body(&child)
 }()
 defer func(){close(scope.cancel);<-finished}()
 var outerCancel <-chan struct{};if m.effects!=nil{outerCancel=m.effects.cancel}
 var drive func() []any
 drive=func() []any {
  for {
   var event effectEvent
   select{case event=<-scope.events:case <-outerCancel:panic(effectCancelled{})}
   if event.done{if event.failure!=nil{panic(event.failure)};return event.result}
   if len(boundary)>0 && boundary[0] && event.target==target {event.target=nil}
   if maskName==event.name && event.target==nil{if maskTarget!=nil{event.target=maskTarget}else{event.skip++}}
   handler,ok:=handlers[event.name]
   if ok{if event.target!=nil{ok=event.target==target}else if event.skip>0{event.skip--;ok=false}}
   if !ok {
    returned:=m.performRoute(event.name,event.args,event.loc,event.never,event.target,event.skip)
    select{case event.reply<-effectReply{returned}:case <-outerCancel:panic(effectCancelled{})}
    continue
   }
   k:=&Continuation{active:true,never:event.never}
   k.resume=func(values []any) []any {
    select{case event.reply<-effectReply{values}:case <-outerCancel:panic(effectCancelled{})}
    return drive()
   }
   return func() []any {defer func(){k.active=false}();return handler(event.args,k)}()
  }
 }
 return drive()
}

func vectorOffset(index int64, size int) int64 {if index<0{return index+int64(size)};return index}
func(m *Machine) indexPosition(index any,size int,vector bool,loc string) int {
 n:=index.(int64)
 if vector{n=vectorOffset(n,size)}
 if n<0||n>=int64(size){m.perform("exn",[]any{"Index out of bounds: "+text(index)},loc,true);fail(loc,"Non-returning effect returned")}
 return int(n)
}
func(m *Machine) listIndex(value,index any,loc string) any {
 if d,ok:=value.(*Dict);ok{v,found:=d.get(index);if !found{m.perform("exn",[]any{"Dictionary key not found: "+text(index)},loc,true);fail(loc,"Non-returning effect returned")};return v}
 if s,ok:=value.(string);ok{return uint8(s[m.indexPosition(index,len(s),false,loc)])}
 xs:=list(value,loc);return xs[m.indexPosition(index,len(xs),true,loc)]
}
func(m *Machine) listOptionIndex(value,index any,loc string) any {
 return m.handle(func(m *Machine) []any{return []any{&ADT{"option","Some",[]any{m.listIndex(value,index,loc)}}}},map[string]EffectHandler{"exn":func(_ []any,_ *Continuation) []any{return []any{&ADT{"option","None",nil}}}},loc)[0]
}
func(m *Machine) indexSetOwned(value,index,replacement any,loc string,reuse bool) any {
 if d,ok:=value.(*Dict);ok{return d.set(index,replacement)}
 if s,ok:=value.(string);ok{i:=m.indexPosition(index,len(s),false,loc);bytes:=[]byte(s);bytes[i]=replacement.(uint8);if !utf8.Valid(bytes){m.perform("exn",[]any{"String update produces invalid UTF-8"},loc,true);fail(loc,"Non-returning effect returned")};return string(bytes)}
 result:=list(value,loc);i:=m.indexPosition(index,len(result),true,loc);if !reuse{result=append([]any{},result...)};result[i]=replacement;return result
}

// Private cancellation is not an effect: user break handlers cannot swallow a take limit.
type iterationStop struct{marker bool}
func iterationBoundary(body func(func())) {
 stop:=&iterationStop{true}
 defer func(){if err:=recover();err!=nil && err!=stop{panic(err)}}()
 body(func(){panic(stop)})
}
func(m *Machine) iterationCount(value any,name,loc string) float64 {
 n:=number(value,loc)
 if math.IsNaN(n)||math.IsInf(n,0)||n<0||n>9007199254740991||math.Trunc(n)!=n {
  m.perform("exn",[]any{name+": count must be a nonnegative safe integer"},loc,true)
  fail(loc,"Non-returning effect returned")
 }
 return n
}

// Empty outer containers can leave the nested element's container type unsolved.
func(m *Machine) walkUnresolved(value any,visit func(any),loc string) {
 switch xs:=value.(type) {
 case []any: for _,item:=range xs{visit(item)}
 case *Range: for i:=0;i<xs.count();i++{m.tick(loc);visit(xs.start+int64(i)*xs.step())}
 case *Collection:for _,item:=range xs.values(){visit(item)}
 case *Dict: xs.walk(func(k,v any){visit(&Tuple{[]any{k,v}})})
 case *Stream: xs.walk(m,visit)
 default: fail(loc,"Unresolved native streem container")
 }
}

type UpdatePart struct {kind,name string;optional bool;index any;cached any;hasCached bool;get func(*Machine,any,any) any;set func(*Machine,any,any,any) any}
func(m *Machine) copyUpdate(value any,path []UpdatePart,replacement any,loc string,reuse bool) any {
 if len(path)==0{return replacement};p:=path[0]
 if p.kind=="field" {
  old:=recordField(value,p.name,loc);r:=value.(*Record);fields:=map[string]any{}
  for k,v:=range r.fields {fields[k]=v};fields[p.name]=m.copyUpdate(old,path[1:],replacement,loc,false);return &Record{fields}
 }
 if p.kind=="tuple" {
  n:=int(number(p.index,loc));old:=tupleIndex(value,n,loc);items:=append([]any(nil),value.(*Tuple).items...)
  items[n]=m.copyUpdate(old,path[1:],replacement,loc,false);return &Tuple{items}
 }
 if len(path)==1{return p.set(m,value,p.index,replacement)}
 old:=p.cached;if !p.hasCached{old=p.get(m,value,p.index)};updated:=m.copyUpdate(old,path[1:],replacement,loc,false);return p.set(m,value,p.index,updated)
}

func(m *Machine) readUpdatePath(value any,path []UpdatePart,loc string) any {
 for i,p:=range path {switch p.kind {case "field":value=recordField(value,p.name,loc);case "tuple":value=tupleIndex(value,int(number(p.index,loc)),loc);default:value=p.get(m,value,p.index);path[i].cached=value;path[i].hasCached=true}}
 return value
}

// Bind the label at invocation, rather than at quotation creation. All named
// operations cross this boundary before ordinary handler selection resumes.
func(m *Machine) labelBlock(body func(*Machine,*EffectLabel) []any,loc string) []any {
 label:=&EffectLabel{}
 return m.handleWith(func(child *Machine) []any{return body(child,label)},nil,loc,label,"",nil,true)
}

func(m *Machine) combinations(xs,other []any,count float64,mode string,visit func(any),loc string){
 if mode=="cartesianProduct"{for _,a:=range xs{for _,b:=range other{m.tick(loc);visit(&Tuple{[]any{a,b}})}};return}
 if mode=="powerset"{for k:=0;k<=len(xs);k++{m.combinations(xs,nil,float64(k),"combinations",visit,loc)};return}
 if count==0{visit([]any{});return};if len(xs)==0||mode!="combinationsWithReplacement"&&count>float64(len(xs)){return}
 // Grow only the live search prefix; no eager allocation proportional to count.
 indices:=[]int{};next:=[]int{0};depth:=0
 for depth>=0{
  m.tick(loc)
  if float64(depth)==count{result:=make([]any,depth);for i:=range result{result[i]=xs[indices[i]]};visit(result);depth--;continue}
  i:=next[depth];next[depth]++
  if i>=len(xs){next=next[:depth];depth--;continue}
  used:=false;if mode=="permutations"{for j:=0;j<depth;j++{if indices[j]==i{used=true;break}}};if used{continue}
  if len(indices)<=depth{indices=append(indices,i)}else{indices[depth]=i};depth++
  start:=0;if mode!="permutations"{start=i;if mode=="combinations"{start++}}
  if len(next)<=depth{next=append(next,start)}else{next[depth]=start}
 }
}

// A zip cursor suspends a source at each item and forwards its effects on demand.
type zipCursor struct{scope *EffectScope;finished chan struct{};pending chan effectReply;done bool;start chan struct{};started bool}
func newZipCursor(m *Machine,walk func(*Machine,func(any)),loc string) *zipCursor {
 c:=&zipCursor{scope:&EffectScope{make(chan effectEvent),make(chan struct{})},finished:make(chan struct{}),start:make(chan struct{})};child:=*m;child.effects=c.scope
 go func(){defer close(c.finished);select{case <-c.start:case <-c.scope.cancel:return};event:=effectEvent{done:true};defer func(){if err:=recover();err!=nil{event.failure=err};select{case c.scope.events<-event:case <-c.scope.cancel:}}();walk(&child,func(item any){child.perform("__zip_item",[]any{item},loc,false)})}()
 return c
}
func(c *zipCursor) close(){close(c.scope.cancel);<-c.finished}
func(c *zipCursor) next(m *Machine) (any,bool) {
 if c.done{return nil,false};if !c.started{c.started=true;close(c.start)};if c.pending!=nil{c.pending<-effectReply{};c.pending=nil}
 for{var event effectEvent;var cancelled <-chan struct{};if m.effects!=nil{cancelled=m.effects.cancel};select{case event=<-c.scope.events:case <-cancelled:panic(effectCancelled{})}
 if event.done{c.done=true;if event.failure!=nil{panic(event.failure)};return nil,false}
 if event.name=="__zip_item"{c.pending=event.reply;return event.args[0],true}
 result:=m.performRoute(event.name,event.args,event.loc,event.never,event.target,event.skip);event.reply<-effectReply{result}
 }
}
func(m *Machine) walkZip(left,right func(*Machine,func(any)),visit func(any),loc string){
 a:=newZipCursor(m,left,loc);defer a.close();b:=newZipCursor(m,right,loc);defer b.close()
 for{x,ok:=a.next(m);if !ok{return};y,ok:=b.next(m);if !ok{return};visit(&Tuple{[]any{x,y}})}
}

// Conditional path updates evaluate replacement commands only at a selected leaf.
func(m *Machine) modifyUpdate(value any,path []UpdatePart,replace func(any) any,defaulting bool,loc string) any {
 if len(path)==0{return replace(value)};p:=path[0];last:=len(path)==1
 if p.kind=="field"{r:=value.(*Record);old:=recordField(value,p.name,loc);if defaulting&&last{return value};updated:=m.modifyUpdate(old,path[1:],replace,defaulting,loc);fields:=map[string]any{};for k,v:=range r.fields{fields[k]=v};fields[p.name]=updated;return &Record{fields}}
 if p.kind=="tuple"{n:=int(number(p.index,loc));old:=tupleIndex(value,n,loc);if defaulting&&last{return value};items:=append([]any{},value.(*Tuple).items...);items[n]=m.modifyUpdate(old,path[1:],replace,defaulting,loc);return &Tuple{items}}
 var old any;found:=true
 if p.optional||defaulting&&last{
  result:=m.handle(func(child *Machine) []any{return []any{true,p.get(child,value,p.index)}},map[string]EffectHandler{"exn":func(_ []any,_ *Continuation) []any{return []any{false,nil}}},loc)
  found=result[0].(bool);old=result[1]
  if !found&&p.optional||found&&defaulting&&last{return value}
 }else{old=p.get(m,value,p.index)}
 updated:=m.modifyUpdate(old,path[1:],replace,defaulting,loc);return p.set(m,value,p.index,updated)
}
func(m *Machine) partitionCount(value any,name,loc string) int64 {
 n,ok:=value.(int64);if !ok||n<=0{m.perform("exn",[]any{name+": size must be a positive integer"},loc,true);fail(loc,"Non-returning effect returned")};return n
}
func(m *Machine) partitionItems(walk func(*Machine,func(any)),key func(any)any,count int64,mode string,visit func(any),loc string){
 pending:=[]any{};var last any;hasKey:=false
 emit:=func(items []any){snapshot:=append([]any{},items...);if mode=="chunkBy"{visit(&Stream{func(m *Machine,v func(any)){for _,x:=range snapshot{v(x)}}})}else{visit(snapshot)}}
 walk(m,func(item any){
  if mode=="chunkBy"{k:=key(item);if hasKey&&!equal(last,k){emit(pending);pending=[]any{}};pending=append(pending,item);last=k;hasKey=true
  }else{pending=append(pending,item);if int64(len(pending))==count{emit(pending);if mode=="windows"{pending=append([]any{},pending[1:]...)}else{pending=[]any{}}}}
 })
 if len(pending)>0&&mode!="windows"{emit(pending)}
}
func(m *Machine) groupItems(walk func(*Machine,func(any)),key func(any)any,result *Dict){
 walk(m,func(item any){k:=key(item);v,found:=result.get(k);items:=[]any{};if found{items=v.([]any)};updated:=result.set(k,append(items,item));*result=*updated})
}
